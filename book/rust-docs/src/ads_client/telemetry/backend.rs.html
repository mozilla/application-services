6//! The telemetry backend: one function per metric in `metrics.yaml`, each
7//! forwarded to the process-wide [`Telemetry`] implementation.
8//!
9//! The function signatures match the glean-sym backend this will be replaced
10//! with, so the swap only touches this file. [`Telemetry`], [`install`] and
11//! [`uninstall`] exist only because the FFI layer hands us a UniFFI callback
12//! that has to be dropped again before shutdown; glean-sym needs none of them.
13//!
14//! Until a callback is installed, and after it is uninstalled, recording is a
15//! no-op.
16
17#[cfg(test)]
18use std::collections::HashMap;
19use std::sync::Arc;
20
21#[cfg(test)]
22use parking_lot::Mutex;
23use parking_lot::RwLock;
24
25use crate::bytesize::ByteSize;

27/// Receives the ads client's metrics.
28pub trait Telemetry: Send + Sync {
29    fn record_build_cache_error(&self, label: &str, value: String);
30    fn record_client_error(&self, label: &str, value: String);
31    fn record_client_operation_total(&self, label: &str);
32    fn record_deserialization_error(&self, label: &str, value: String);
33    fn record_http_cache_outcome(&self, label: &str, value: String);
34}
35
36// By design: telemetry is a process-wide singleton, as it will be with glean-sym.
37static CALLBACK: RwLock<Option<Arc<dyn Telemetry>>> = RwLock::new(None);
38
39/// Makes `callback` the destination of every metric recorded from now on.
40pub(crate) fn install(callback: impl Telemetry + 'static) {
41    // Dropping a UniFFI callback calls into foreign code, so let the replaced
42    // one drop after the lock is released.
43    let _replaced = CALLBACK.write().replace(Arc::new(callback));
44}
45
46/// Drops the installed callback, after which recording is a no-op.
47///
48/// UniFFI callbacks left hanging at shutdown crash Firefox Desktop on quit, so
49/// the client uninstalls the callback when it shuts down.
50pub(crate) fn uninstall() {
51    // As in `install`, the removed callback drops after the lock is released.
52    let _removed = CALLBACK.write().take();
53}
54
55fn with_callback(f: impl FnOnce(&dyn Telemetry)) {
56    // Clone out of the lock so a slow callback never holds up `install` or
57    // `uninstall`.
58    let Some(callback) = CALLBACK.read().clone() else {
59        return;
60    };
61    f(callback.as_ref());
62}

64/// Glean drops the end of longer string values and records an
65/// `invalid_overflow` error (bug 2047640).
66const MAX_STRING_SIZE: ByteSize = ByteSize::b(255);
67
68/// Cuts `value` to fit in a Glean string, on a char boundary.
69fn fit(mut value: String) -> String {
70    let max = usize::try_from(MAX_STRING_SIZE.as_u64()).unwrap_or(usize::MAX);
71    if value.len() > max {
72        // `str::floor_char_boundary` does this, but needs Rust 1.91 and
73        // mozilla-central still builds with 1.90.
74        let mut end = max;
75        while !value.is_char_boundary(end) {
76            end -= 1;
77        }
78        value.truncate(end);
79    }
80    value
81}
82
83pub(super) fn build_cache_error(label: &str, value: String) {
84    with_callback(|cb| cb.record_build_cache_error(label, fit(value)));
85}
86
87pub(super) fn client_error(label: &str, value: String) {
88    with_callback(|cb| cb.record_client_error(label, fit(value)));
89}
90
91pub(super) fn client_operation_total(label: &str) {
92    with_callback(|cb| cb.record_client_operation_total(label));
93}
94
95pub(super) fn deserialization_error(label: &str, value: String) {
96    with_callback(|cb| cb.record_deserialization_error(label, fit(value)));
97}
98
99pub(super) fn http_cache_outcome(label: &str, value: String) {
100    with_callback(|cb| cb.record_http_cache_outcome(label, fit(value)));
101}

103/// Serializes tests that install a callback, since they all share the one
104/// singleton.
105#[cfg(test)]
106pub fn test_lock() -> parking_lot::MutexGuard<'static, ()> {
107    static LOCK: parking_lot::Mutex<()> = parking_lot::Mutex::new(());
108    LOCK.lock()
109}

111#[cfg(test)]
112#[derive(Clone, Default)]
113pub struct RecordingTelemetry {
114    pub build_cache_errors: Arc<Mutex<HashMap<String, Vec<String>>>>,
115    pub client_errors: Arc<Mutex<HashMap<String, Vec<String>>>>,
116    pub client_operation_totals: Arc<Mutex<HashMap<String, u32>>>,
117    pub deserialization_errors: Arc<Mutex<HashMap<String, Vec<String>>>>,
118    pub http_cache_outcomes: Arc<Mutex<HashMap<String, Vec<String>>>>,
119}
120
121#[cfg(test)]
122impl Telemetry for RecordingTelemetry {
123    fn record_build_cache_error(&self, label: &str, value: String) {
124        self.build_cache_errors
125            .lock()
126            .entry(label.to_string())
127            .or_default()
128            .push(value);
129    }
130    fn record_client_error(&self, label: &str, value: String) {
131        self.client_errors
132            .lock()
133            .entry(label.to_string())
134            .or_default()
135            .push(value);
136    }
137    fn record_client_operation_total(&self, label: &str) {
138        *self
139            .client_operation_totals
140            .lock()
141            .entry(label.to_string())
142            .or_default() += 1;
143    }
144    fn record_deserialization_error(&self, label: &str, value: String) {
145        self.deserialization_errors
146            .lock()
147            .entry(label.to_string())
148            .or_default()
149            .push(value);
150    }
151    fn record_http_cache_outcome(&self, label: &str, value: String) {
152        self.http_cache_outcomes
153            .lock()
154            .entry(label.to_string())
155            .or_default()
156            .push(value);
157    }
158}

160#[cfg(test)]
161mod tests {
162    use super::*;
163    use crate::telemetry::{record_client_operation, ClientOperation};
164
165    #[test]
166    fn test_fit_cuts_on_a_char_boundary() {
167        assert_eq!(fit("short".into()), "short");
168        assert_eq!(fit("a".repeat(300)).len() as u64, MAX_STRING_SIZE.as_u64());
169        let cut = fit("é".repeat(200));
170        assert_eq!(cut.len(), 254);
171        assert!(cut.chars().all(|c| c == 'é'));
172    }
173
174    #[test]
175    fn test_records_to_installed_callback() {
176        let _lock = test_lock();
177        let recording = RecordingTelemetry::default();
178        install(recording.clone());
179
180        record_client_operation(ClientOperation::ReportAd);
181        // Other tests may record concurrently, so only look for ours.
182        assert!(recording
183            .client_operation_totals
184            .lock()
185            .keys()
186            .any(|label| label == "report_ad"));
187
188        uninstall();
189        recording.client_operation_totals.lock().clear();
190        record_client_operation(ClientOperation::ReportAd);
191        assert!(recording.client_operation_totals.lock().is_empty());
192    }
