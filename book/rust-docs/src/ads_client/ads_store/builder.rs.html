5use super::connection_initializer::AdsStoreConnectionInitializer;
6use crate::ads_store::store::AdsStoreHolder;
7use crate::ads_store::AdsStore;
8use crate::common::bytesize::ByteSize;
9use rusqlite::Connection;
10use sql_support::open_database;
11use std::path::PathBuf;
12
13const DEFAULT_MAX_SIZE: ByteSize = ByteSize::mib(10);
14const MIN_STORE_SIZE: ByteSize = ByteSize::kib(1);
15const MAX_STORE_SIZE: ByteSize = ByteSize::mib(100);
16
17#[derive(Debug, thiserror::Error)]
18pub enum AdsStoreBuilderError {
19    #[error("Database error: {0}")]
20    Database(#[from] open_database::Error),
21    #[error("Database path cannot be empty")]
22    EmptyDbPath,
23    #[error(
24        "Maximum store size must be between {min_size} and {max_size}, got {size_bytes} bytes"
25    )]
26    InvalidMaxSize {
27        max_size: String,
28        min_size: String,
29        size_bytes: u64,
30    },
31}
32
33pub struct AdsStoreBuilder {
34    db_path: PathBuf,
35    max_size: Option<ByteSize>,
36}
37
38impl AdsStoreBuilder {
39    pub fn new(db_path: impl Into<PathBuf>) -> Self {
40        Self {
41            db_path: db_path.into(),
42            max_size: None,
43        }
44    }
45
46    pub fn max_size(mut self, max_size: ByteSize) -> Self {
47        self.max_size = Some(max_size);
48        self
49    }
50
51    fn open_connection(&self) -> Result<Connection, AdsStoreBuilderError> {
52        let initializer = AdsStoreConnectionInitializer {};
53        let conn = if cfg!(test) {
54            open_database::open_memory_database(&initializer)?
55        } else {
56            open_database::open_database(&self.db_path, &initializer)?
57        };
58        Ok(conn)
59    }
60
61    fn validate(&self) -> Result<(), AdsStoreBuilderError> {
62        if self.db_path.to_string_lossy().trim().is_empty() {
63            return Err(AdsStoreBuilderError::EmptyDbPath);
64        }
65
66        if let Some(max_size) = self.max_size {
67            if max_size < MIN_STORE_SIZE || max_size > MAX_STORE_SIZE {
68                return Err(AdsStoreBuilderError::InvalidMaxSize {
69                    size_bytes: max_size.as_u64(),
70                    min_size: MIN_STORE_SIZE.to_string(),
71                    max_size: MAX_STORE_SIZE.to_string(),
72                });
73            }
74        }
75
76        Ok(())
77    }
78
79    pub fn build(&self) -> Result<AdsStore, AdsStoreBuilderError> {
80        self.validate()?;
81
82        let conn = self.open_connection()?;
83        let holder = AdsStoreHolder::new(conn);
84        let max_size = self.max_size.unwrap_or(DEFAULT_MAX_SIZE);
85        Ok(AdsStore { max_size, holder })
86    }
87}

89#[cfg(test)]
90mod tests {

93    fn make_test_builder(path: &str) -> AdsStoreBuilder {
94        AdsStoreBuilder::new(path)

97    #[test]
98    fn test_store_builder_with_defaults() {
99        let builder = make_test_builder("test.db");
100        assert_eq!(builder.db_path, PathBuf::from("test.db"));
101        assert_eq!(builder.max_size, None);
102        assert!(builder.build().is_ok());
103    }
104
105    #[test]
106    fn test_cache_builder_valid_custom() {
107        let builder = make_test_builder("custom.db").max_size(ByteSize::b(1024));
108
109        assert_eq!(builder.db_path, PathBuf::from("custom.db"));
110        assert_eq!(builder.max_size, Some(ByteSize::b(1024)));
111        assert!(builder.build().is_ok());
112    }
113
114    #[test]
115    fn test_validation_empty_db_path() {
116        let result = make_test_builder("   ").build();
117        assert!(matches!(result, Err(AdsStoreBuilderError::EmptyDbPath)));
118    }
119
120    #[test]
121    fn test_validation_max_size_too_small() {
122        let result = make_test_builder("test.db")
123            .max_size(ByteSize::b(512))
124            .build();
125        assert!(matches!(
126            result,
127            Err(AdsStoreBuilderError::InvalidMaxSize {
128                size_bytes: 512,
129                min_size: _,
130                max_size: _,
131            })
132        ));
133    }

135    #[test]
136    fn test_validation_max_size_too_large() {
137        let result = make_test_builder("test.db")
138            .max_size(ByteSize::b(2 * 1024 * 1024 * 1024))
139            .build();

141            result,
142            Err(AdsStoreBuilderError::InvalidMaxSize {
143                size_bytes: 2147483648,
144                min_size: _,
145                max_size: _,

147        ));
148    }

150    #[test]

152        let builder_min = make_test_builder("test.db").max_size(MIN_STORE_SIZE);
