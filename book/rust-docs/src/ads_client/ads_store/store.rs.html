40    /// Removes all entries from cache.
41    pub fn clear_all(&self) -> Result<usize, open_database::Error> {
42        let (conn, _) = self.db.lock()?;
43        let mut total = 0;
44        total += conn.execute("DELETE FROM ads", [])?;
45        Ok(total)
46    }
47
48    pub fn close(&self) {
49        self.db.close(true);
50    }

52    /// Returns whether the db is in memory rather than backed by a file.
53    pub fn is_memory(&self) -> Result<bool, open_database::Error> {
54        let (conn, _) = self.db.lock()?;
55        // SQLite reports an empty path for in-memory dbs.
56        Ok(conn.path() == Some(""))
57    }

59    /// Returns total size of the cache in bytes.
60    pub fn current_total_size_bytes(&self) -> Result<ByteSize, open_database::Error> {
61        let (conn, _) = self.db.lock()?;
62        let size_bytes_ads: u64 =
63            conn.query_row("SELECT COALESCE(SUM(size_bytes),0) FROM ads", [], |row| {
64                row.get(0)
65            })?;
66        Ok(ByteSize::b(size_bytes_ads))
67    }

80    pub fn lookup(&self, placement_id: &PlacementId) -> Result<Option<Vec<Ad>>, FetchAdsError> {
81        #[cfg(test)]
82        if *self.fault.lock() == FaultKind::Lookup {
83            return Err(open_database::Error::from(Self::forced_fault_error(
84                "forced lookup failure",
85            )))?;
86        }
87        let (conn, _) = self.db.lock()?;
88        let res = conn
89            .query_row(
90                "SELECT placement_id, ad_body
91                 FROM ads
92                 WHERE placement_id = ?1",
93                params![placement_id.as_ref()],
94                |row| {
95                    let ad_body: Vec<u8> = row.get(1)?;
96                    Ok(ad_body)
97                },
98            )
99            .optional()
100            .map_err(open_database::Error::from)?;
101        Ok(res.map(|x| serde_json::from_slice(&x)).transpose()?)
102    }
103
104    #[cfg(test)]
105    pub fn set_fault(&self, kind: FaultKind) {
106        *self.fault.lock() = kind;
107    }

109    // Upsert multiple ads for a singular placement, replacing any ads already stored for it.
110    pub fn store_ads(&self, placement_id: &PlacementId, ads: &[Ad]) -> Result<(), FetchAdsError> {
111        self.store_batch_ads([(placement_id, ads)])
112    }

114    /// Upsert multiple ads for many placements, replacing any ads already stored for it.
115    pub fn store_batch_ads<'a, K: AsRef<str>, V: AsRef<[Ad]>>(
116        &self,
117        ads_map: impl IntoIterator<Item = (K, V)>,
118    ) -> Result<(), FetchAdsError> {
119        #[cfg(test)]
120        if *self.fault.lock() == FaultKind::Store {
121            return Err(open_database::Error::from(Self::forced_fault_error(
122                "forced store failure",
123            )))?;
124        }
125
126        let (mut conn, _) = self.db.lock()?;
127        let tx = conn.transaction().map_err(open_database::Error::from)?;
128        let mut stmt = tx
129            .prepare(
130                "INSERT INTO ads (
131                stored_at,
132                placement_id,
133                ad_body,
134                size_bytes
135            )
136            VALUES (?1, ?2, ?3, ?4)
137            ON CONFLICT(placement_id) DO UPDATE SET
138                stored_at=excluded.stored_at,
139                ad_body=excluded.ad_body,
140                size_bytes=excluded.size_bytes",
141            )
142            .map_err(open_database::Error::from)?;
143
144        let now = self.clock.now();
145        for (placement_id, ads) in ads_map {
146            let placement_id_str: &str = placement_id.as_ref();
147            let ads: &[Ad] = ads.as_ref();
148            let ad_body = serde_json::to_vec(&ads)?;
149            let size_bytes = ad_body.len();
150            stmt.execute(params![
151                now.timestamp(),
152                placement_id_str,
153                ad_body,
154                size_bytes,
155            ])
156            .map_err(open_database::Error::from)?;
157        }
158        stmt.finalize().map_err(open_database::Error::from)?;
159        tx.commit().map_err(open_database::Error::from)?;
160
161        Ok(())
162    }

164    pub fn trim_to_max_size(&self, max_size: &ByteSize) -> Result<(), open_database::Error> {
165        #[cfg(test)]
166        if *self.fault.lock() == FaultKind::Trim {
167            return Err(Self::forced_fault_error("forced trim failure"))?;
168        }
169        // Keeps the newest rows that fit in `max_size` and deletes the rest,
170        // in one statement.
171        let (conn, _) = self.db.lock()?;
172        conn.execute(
173            "DELETE FROM ads
174             WHERE rowid IN (
175                 SELECT rowid
176                 FROM (
177                     SELECT rowid,
178                            SUM(size_bytes) OVER (ORDER BY stored_at DESC, rowid DESC) AS kept_bytes
179                     FROM ads
180                 )
181                 WHERE kept_bytes > ?1
182             )",
183            params![max_size.as_u64()],
184        )?;
185        Ok(())
186    }

188    #[cfg(test)]
189    fn forced_fault_error(msg: &str) -> rusqlite::Error {
190        rusqlite::Error::SqliteFailure(
191            rusqlite::ffi::Error {
192                code: rusqlite::ErrorCode::InternalMalfunction,
193                extended_code: 0,
194            },
195            Some(msg.to_string()),
196        )
197    }
