5use chrono::{DateTime, Utc};
6use std::time::Duration;
7use viaduct::{header_names, Response};
8
9/// How old a response already was when it was received, from its `Age` and
10/// `Date` headers (RFC 9111 §4.2.3). A missing or unreadable header counts
11/// as zero.
12pub(super) struct ResponseAge {
13    age: Duration,
14    date: Option<DateTime<Utc>>,
15}
16
17impl From<&Response> for ResponseAge {
18    fn from(response: &Response) -> Self {
19        Self {
20            age: response
21                .headers
22                .get(header_names::AGE)
23                .and_then(|value| value.trim().parse().ok())
24                .map(Duration::from_secs)
25                .unwrap_or_default(),
26            date: response
27                .headers
28                .get(header_names::DATE)
29                .and_then(|value| DateTime::parse_from_rfc2822(value.trim()).ok())
30                .map(|date| date.to_utc()),
31        }
32    }
33}
34
35impl ResponseAge {
36    /// The larger of `Age` and the time elapsed since `Date`, as of `now`.
37    pub(super) fn at(&self, now: DateTime<Utc>) -> Duration {
38        let since_date = self
39            .date
40            .and_then(|date| (now - date).to_std().ok())
41            .unwrap_or_default();
42        self.age.max(since_date)
43    }
44}

46#[cfg(test)]
47mod tests {
48    use super::*;
49    use viaduct::{Headers, Method};
50
51    fn age_at_now(headers: &[(&str, &str)]) -> Duration {
52        let mut map = Headers::new();
53        for (name, value) in headers {
54            map.insert(name.to_string(), *value).unwrap();
55        }
56        let response = Response {
57            body: vec![],
58            headers: map,
59            request_method: Method::Get,
60            status: 200,
61            url: "https://example.com".parse().unwrap(),
62        };
63        let now = "2025-11-06T08:49:37Z".parse().unwrap();
64        ResponseAge::from(&response).at(now)
65    }
66
67    #[test]
68    fn no_age_or_date_is_zero() {
69        assert_eq!(age_at_now(&[]), Duration::ZERO);
70    }
71
72    #[test]
73    fn uses_age_header() {
74        assert_eq!(age_at_now(&[("age", "72000")]), Duration::from_secs(72000));
75    }
76
77    #[test]
78    fn uses_time_since_date_header() {
79        assert_eq!(
80            age_at_now(&[("date", "Thu, 06 Nov 2025 07:49:37 GMT")]),
81            Duration::from_secs(3600)
82        );
83    }
84
85    #[test]
86    fn takes_the_larger_of_age_and_date() {
87        let date = ("date", "Thu, 06 Nov 2025 07:49:37 GMT");
88        assert_eq!(
89            age_at_now(&[("age", "60"), date]),
90            Duration::from_secs(3600)
91        );
92        assert_eq!(
93            age_at_now(&[("age", "7200"), date]),
94            Duration::from_secs(7200)
95        );
96    }
97
98    #[test]
99    fn date_in_the_future_is_zero() {
100        assert_eq!(
101            age_at_now(&[("date", "Thu, 06 Nov 2025 09:49:37 GMT")]),
102            Duration::ZERO
103        );
104    }
105
106    #[test]
107    fn unreadable_headers_are_ignored() {
108        assert_eq!(
109            age_at_now(&[("age", "-5"), ("date", "yesterday")]),
110            Duration::ZERO
111        );
112    }
113}
