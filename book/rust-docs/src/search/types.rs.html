10/// The list of possible application names that are currently supported.
11#[derive(Clone, Debug, Default, Deserialize, PartialEq, uniffi::Enum)]

13pub enum SearchApplicationName {
14    FirefoxAndroid = 1,
15    FirefoxIos = 2,
16    FocusAndroid = 3,
17    FocusIos = 4,
18    // The default doesn't really matter here, so we pick desktop.
19    #[default]
20    Firefox = 5,
21}

25        match self {
26            SearchApplicationName::Firefox => "firefox",
27            SearchApplicationName::FirefoxAndroid => "firefox-android",
28            SearchApplicationName::FocusAndroid => "focus-android",
29            SearchApplicationName::FirefoxIos => "firefox-ios",
30            SearchApplicationName::FocusIos => "focus-ios",
31        }
32    }
33}

181/// Parameter definitions for search engine URLs. The name property is always
182/// specified, along with one of value, experiment_config or search_access_point.
183#[derive(Debug, uniffi::Record, PartialEq, Deserialize, Clone)]
184#[serde(rename_all = "camelCase")]
185pub struct SearchUrlParam {
186    /// The name of the parameter in the url.
187    pub name: String,
188
189    /// The parameter value, this may be a static value, or additionally contain
190    /// a parameter replacement, e.g. `{inputEncoding}`. For the partner code
191    /// parameter, this field should be `{partnerCode}`.
192    pub value: Option<String>,
193
194    /// Same as value but only used if Services.polices.isEnterprise is true. Overrides other parameters of the same name.
195    pub enterprise_value: Option<String>,
196
197    /// The value for the parameter will be derived from the equivalent experiment
198    /// configuration value.
199    /// Only desktop uses this currently.
200    pub experiment_config: Option<String>,
201}

203/// Defines an individual search engine URL.
204#[derive(Debug, uniffi::Record, PartialEq, Deserialize, Clone)]
205pub struct SearchEngineUrl {
206    /// The PrePath and FilePath of the URL. May include variables for engines
207    /// which have a variable FilePath, e.g. `{searchTerms}` for when a search
208    /// term is within the path of the url.
209    pub base: String,
210
211    /// The HTTP method to use to send the request (`GET` or `POST`).
212    /// If the engine definition has not specified the method, it defaults to GET.
213    pub method: String,
214
215    /// The parameters for this URL.
216    pub params: Vec<SearchUrlParam>,
217
218    /// The name of the query parameter for the search term. Automatically
219    /// appended to the end of the query. This may be skipped if `{searchTerms}`
220    /// is included in the base.
221    pub search_term_param_name: Option<String>,
222
223    /// The display name of the URL, if any. This is useful if the URL
224    /// corresponds to a brand name distinct from the engine's brand name.
225    #[uniffi(default = None)]
226    pub display_name: Option<String>,
227
228    /// Indicates the date until which the URL is considered new
229    /// (format: YYYY-MM-DD).
230    #[uniffi(default = None)]
231    pub is_new_until: Option<String>,
232
233    /// Whether the engine's partner code should be excluded from telemetry when
234    /// this URL is visited.
235    #[uniffi(default = false)]
236    pub exclude_partner_code_from_telemetry: bool,
237
238    /// If this URL performs searches only for certain MIME types, they should
239    /// be listed here. If `None`, it's assumed the content type is text or not
240    /// relevant. This field is intended to be used for URLs like visual search,
241    /// which might support certain image types and not others. Consumers can
242    /// use it to determine whether search UI corresponding to the URL should be
243    /// shown to the user in a given context.
244    #[uniffi(default = None)]
245    pub accepted_content_types: Option<Vec<String>>,
246}

307/// A definition for an individual search engine to be presented to the user.
308#[derive(Debug, uniffi::Record, PartialEq, Clone, Default)]
309pub struct SearchEngineDefinition {
310    /// A list of aliases for this engine.
311    pub aliases: Vec<String>,
312
313    /// The character set this engine uses for queries.
314    pub charset: String,
315
316    /// The classification of search engine according to the main search types
317    /// (e.g. general, shopping, travel, dictionary). Currently, only marking as
318    /// a general search engine is supported.
319    /// On Android, only general search engines may be selected as "default"
320    /// search engines.
321    pub classification: SearchEngineClassification,
322
323    /// The identifier of the search engine. This is used as an internal
324    /// identifier, e.g. for saving the user's settings for the engine. It is
325    /// also used to form the base telemetry id and may be extended by telemetrySuffix.
326    pub identifier: String,
327
328    /// Indicates the date until which the engine variant or subvariant is considered new
329    /// (format: YYYY-MM-DD).
330    pub is_new_until: Option<String>,
331
332    /// The user visible name of the search engine.
333    pub name: String,
334
335    /// This search engine is presented as an option that the user may enable.
336    /// The application should not include these in the default list of the
337    /// user's engines. If not supported, it should filter them out.
338    pub optional: bool,
339
340    /// The partner code for the engine. This will be inserted into parameters
341    /// which include `{partnerCode}`. May be the empty string.
342    pub partner_code: String,
343
344    /// Optional suffix that is appended to the search engine identifier
345    /// following a dash, i.e. `<identifier>-<suffix>`. If it is an empty string
346    /// no dash should be appended.
347    pub telemetry_suffix: String,
348
349    /// The URLs associated with the search engine.
350    pub urls: SearchEngineUrls,
351
352    /// A hint to the order that this engine should be in the engine list. This
353    /// is derived from the `engineOrders` section of the search configuration.
354    /// The higher the number, the nearer to the front it should be.
355    /// If the number is not specified, other methods of sorting may be relied
356    /// upon (e.g. alphabetical).
357    pub order_hint: Option<u32>,
358
359    /// The url used for reporting clicks.
360    pub click_url: Option<String>,
361}

426/// Details of the search engines to display to the user, generated as a result
427/// of processing the search configuration.
428#[derive(Debug, uniffi::Record, PartialEq)]
429pub struct RefinedSearchConfig {
430    /// A sorted list of engines. Clients may use the engine in the order that
431    /// this list is specified, or they may implement their own order if they
432    /// have other requirements.
433    ///
434    /// The application default engines should not be assumed from this order in
435    /// case of future changes.
436    ///
437    /// The sort order is:
438    ///
439    /// * Application Default Engine
440    /// * Application Default Engine for Private Mode (if specified & different)
441    /// * Engines sorted by descending `SearchEngineDefinition.orderHint`
442    /// * Any other engines in alphabetical order (locale based comparison)
443    pub engines: Vec<SearchEngineDefinition>,
444
445    /// The identifier of the engine that should be used for the application
446    /// default engine. If this is undefined, an error has occurred, and the
447    /// application should either default to the first engine in the engines
448    /// list or otherwise handle appropriately.
449    pub app_default_engine_id: Option<String>,
450
451    /// If specified, the identifier of the engine that should be used for the
452    /// application default engine in private browsing mode.
453    /// Only desktop uses this currently.
454    pub app_private_default_engine_id: Option<String>,
455}

488#[cfg(test)]
489mod tests {
490    use super::*;
491    use serde_json::json;

493    #[test]
494    fn test_search_application_name_v3_deserializes() {
495        for (value, expected) in [
496            ("firefox-win", SearchApplicationNameV3::FirefoxWin),
497            ("firefox-macosx", SearchApplicationNameV3::FirefoxMacosx),
498            ("firefox-linux", SearchApplicationNameV3::FirefoxLinux),
499            ("firefox-android", SearchApplicationNameV3::FirefoxAndroid),
500            ("firefox-ios", SearchApplicationNameV3::FirefoxIos),
501            ("focus-android", SearchApplicationNameV3::FocusAndroid),
502            ("focus-ios", SearchApplicationNameV3::FocusIos),
503        ] {
504            let result: SearchApplicationNameV3 = serde_json::from_value(json!(value)).unwrap();
505            assert_eq!(result, expected, "Should deserialize {value}");
506            assert_eq!(result.as_str(), value, "Should round trip {value}");
507        }
508    }
509
510    #[test]
511    fn test_v3_enums_fall_back_for_unknown_values() {
512        let app_name: SearchApplicationNameV3 = serde_json::from_value(json!("firefox")).unwrap();
513        assert_eq!(app_name, SearchApplicationNameV3::Unknown);
514
515        let channel: SearchUpdateChannelV3 = serde_json::from_value(json!("unknown")).unwrap();
516        assert_eq!(channel, SearchUpdateChannelV3::Default);
517
518        let device_type: SearchDeviceTypeV3 = serde_json::from_value(json!("unknown")).unwrap();
519        assert_eq!(device_type, SearchDeviceTypeV3::None);
520
521        let classification: SearchEngineClassificationV3 =
522            serde_json::from_value(json!("unknown-type")).unwrap();
523        assert_eq!(classification, SearchEngineClassificationV3::Unknown);
524    }
525
526    #[test]
527    fn test_v2_enums_do_not_fall_back_for_unknown_values() {
528        assert!(serde_json::from_value::<SearchApplicationName>(json!("firefox-win")).is_err());
529        assert!(serde_json::from_value::<SearchUpdateChannel>(json!("unknown")).is_err());
530        assert!(serde_json::from_value::<SearchDeviceType>(json!("unknown")).is_err());
531        assert!(
532            serde_json::from_value::<SearchEngineClassification>(json!("unknown-type")).is_err()
533        );
534    }
