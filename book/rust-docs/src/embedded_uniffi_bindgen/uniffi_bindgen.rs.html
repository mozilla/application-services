14/// TargetLanguage uniffi_bindgen, with a `clap::ValueEnum` derive.
15#[derive(Copy, Clone, ValueEnum)]
16enum TargetLanguageArg {
17    Kotlin,
18    Swift,
19    Python,
20    Ruby,
21}
22
23impl fmt::Display for TargetLanguageArg {
24    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
25        match self {
26            Self::Kotlin => write!(f, "kotlin"),
27            Self::Swift => write!(f, "swift"),
28            Self::Python => write!(f, "python"),
29            Self::Ruby => write!(f, "ruby"),
30        }
31    }
32}
33
34impl From<TargetLanguageArg> for TargetLanguage {
35    fn from(l: TargetLanguageArg) -> Self {
36        match l {
37            TargetLanguageArg::Kotlin => Self::Kotlin,
38            TargetLanguageArg::Swift => Self::Swift,
39            TargetLanguageArg::Python => Self::Python,
40            TargetLanguageArg::Ruby => Self::Ruby,
41        }
42    }
43}

48/// Scaffolding and bindings generator for Rust
49#[derive(Parser)]
50#[clap(name = "uniffi-bindgen")]
51struct Cli {
52    #[clap(subcommand)]
53    command: Commands,
54}
55
56#[derive(Subcommand)]
57enum Commands {
58    /// Generate foreign language bindings
59    Generate {
60        /// Foreign language(s) for which to build bindings.
61        #[clap(long, short, value_enum)]
62        language: Vec<TargetLanguageArg>,
63
64        /// Directory in which to write generated files. Default is same folder as .udl file.
65        #[clap(long, short)]
66        out_dir: Option<Utf8PathBuf>,
67
68        /// Do not try to format the generated bindings.
69        #[clap(long, short)]
70        no_format: bool,
71
72        /// Path to optional uniffi config file. This config is merged with the `uniffi.toml` config present in each crate, with its values taking precedence.
73        #[clap(long, short)]
74        config: Option<Utf8PathBuf>,
75
76        /// When `--library` is passed, only generate bindings for one crate.
77        /// When `--library` is not passed, use this as the crate name instead of attempting to
78        /// locate and parse Cargo.toml.
79        #[clap(long = "crate")]
80        crate_name: Option<String>,
81
82        /// Path to the UDL file, or cdylib if `library-mode` is specified
83        source: Utf8PathBuf,
84    },
85
86    /// Generate Rust scaffolding code
87    Scaffolding {
88        /// Directory in which to write generated files. Default is same folder as .udl file.
89        #[clap(long, short)]
90        out_dir: Option<Utf8PathBuf>,
91
92        /// Do not try to format the generated bindings.
93        #[clap(long, short)]
94        no_format: bool,
95
96        /// Path to the UDL file.
97        udl_file: Utf8PathBuf,
98    },
99
100    /// Inspect the bindings render pipeline
101    Pipeline(PipelineArgs),
102}
103
104#[derive(Args)]
105struct PipelineArgs {
106    /// Pass in a cdylib path rather than a UDL file
107    #[clap(long = "library")]
108    library_mode: bool,
109
110    /// Path to the UDL file, or cdylib if `library-mode` is specified
111    source: Utf8PathBuf,
112
113    /// When `--library` is passed, only generate bindings for one crate.
114    /// When `--library` is not passed, use this as the crate name instead of attempting to
115    /// locate and parse Cargo.toml.
116    #[clap(long = "crate")]
117    crate_name: Option<String>,
118
119    /// Whether we should exclude dependencies when running "cargo metadata".
120    /// This will mean external types may not be resolved if they are implemented in crates
121    /// outside of this workspace.
122    /// This can be used in environments when all types are in the namespace and fetching
123    /// all sub-dependencies causes obscure platform specific problems.
124    #[clap(long)]
125    metadata_no_deps: bool,
126
127    /// Bindings Language
128    language: TargetLanguageArg,
129
130    /// Only show passes that match <PASS>
131    ///
132    /// Use `last` to only show the last pass, this can be useful when you're writing new pipelines
133    #[clap(short, long)]
134    pass: Option<String>,
135
136    /// Don't show diffs for middle passes
137    #[clap(long)]
138    no_diff: bool,
139
140    /// Only show data for types with name <FILTER_TYPE>
141    #[clap(short = 't', long = "type")]
142    filter_type: Option<String>,
143
144    /// Only show data for items with fields that match <FILTER>
145    #[clap(short = 'n', long = "name")]
146    filter_name: Option<String>,
147}

149pub fn run_main() -> anyhow::Result<()> {
150    let cli = Cli::parse();
151    let metadata_no_deps = false;
152    match cli.command {
153        Commands::Generate {
154            language,
155            out_dir,
156            no_format,
157            config,
158            source,
159            crate_name,
160            ..
161        } => {
162            if language.is_empty() {
163                panic!("please specify at least one language with --language")
164            }
165
166            let mut paths = uniffi_bindgen::BindgenPaths::default();
167            paths.add_layer(crate::config_supplier::NoCargoConfigSupplier);
168
169            generate_with_bindgen_paths(
170                GenerateOptions {
171                    languages: language.into_iter().map(TargetLanguage::from).collect(),
172                    out_dir: out_dir
173                        .expect("--out-dir is required when generating {language} bindings"),
174                    source,
175                    config_override: config,
176                    crate_filter: crate_name,
177                    metadata_no_deps,
178                    format: !no_format,
179                },
180                paths,
181            )?;
182        }
183        Commands::Scaffolding {
184            out_dir,
185            no_format,
186            udl_file,
187        } => {
188            uniffi_bindgen::generate_component_scaffolding(
189                &udl_file,
190                out_dir.as_deref(),
191                !no_format,
192            )?;
193        }
194        Commands::Pipeline(args) => {
195            let mut paths = uniffi_bindgen::BindgenPaths::default();
196            paths.add_layer(crate::config_supplier::NoCargoConfigSupplier);
197            let config = uniffi_bindgen::GlobalConfig::default();
198            let loader = BindgenLoader::new(paths, config);
199            let metadata = loader.load_metadata(&args.source)?;
200            let initial_root = loader.load_pipeline_initial_root(&args.source, metadata)?;
201
202            let opts = PrintOptions {
203                pass: args.pass,
204                no_diff: args.no_diff,
205                filter_type: args.filter_type,
206                filter_name: args.filter_name,
207            };
208            match args.language {
209                TargetLanguageArg::Python => python::pipeline().print_passes(initial_root, opts)?,
210                language => unimplemented!("{language} does not use the bindings IR pipeline yet"),
211            };
212        }
213    };
214    Ok(())
215}
