30/// `ColorChoice::Auto` only looks at `TERM`, not at whether stdout is a terminal, so
31/// on its own it writes escape codes into redirected output and CI logs.
32fn stdout_stream() -> StandardStream {
33    let choice = if std::io::stdout().is_terminal() {
34        ColorChoice::Auto
35    } else {
36        ColorChoice::Never
37    };
38    StandardStream::stdout(choice)
39}

41pub(crate) fn generate_struct(cmd: &GenerateStructCmd) -> Result<()> {
42    let files: FileLoader = TryFrom::try_from(&cmd.loader)?;
43
44    let filename = &cmd.manifest;
45    let input = files.file_path(filename)?;
46
47    validate(&ValidateCmd {
48        manifest: cmd.manifest.clone(),
49        loader: cmd.loader.clone(),
50    })?;
51
52    match (&input, &cmd.output.is_dir()) {
53        (FilePath::Remote(_), _) => generate_struct_single(&files, input, cmd),
54        (FilePath::Local(file), _) if !file.exists() => Err(FMLError::CliError(format!(
55            "Input file or directory `{}' does not exist",
56            filename
57        ))),
58        (FilePath::Local(file), _) if file.is_file() => generate_struct_single(&files, input, cmd),
59        (FilePath::Local(dir), true) if dir.is_dir() => generate_struct_from_dir(&files, cmd, dir),
60        (_, true) => generate_struct_from_glob(&files, cmd, filename),
61        _ => Err(FMLError::CliError(
62            "Cannot generate a single output file from an input directory".to_string(),
63        )),
64    }
65}

67fn generate_struct_from_dir(files: &FileLoader, cmd: &GenerateStructCmd, cwd: &Path) -> Result<()> {
68    let entries = cwd.read_dir()?;
69    for entry in entries.filter_map(Result::ok) {
70        let pb = entry.path();
71        if pb.is_dir() {
72            generate_struct_from_dir(files, cmd, &pb)?;
73        } else if let Some(nm) = pb.file_name().map(|s| s.to_str().unwrap_or_default()) {
74            if nm.ends_with(MATCHING_FML_EXTENSION) {
75                let path = pb.as_path().into();
76                generate_struct_single(files, path, cmd)?;
77            }
78        }
79    }
80    Ok(())
81}

83fn generate_struct_from_glob(
84    files: &FileLoader,
85    cmd: &GenerateStructCmd,
86    pattern: &str,
87) -> Result<()> {
88    use glob::glob_with;
89    let entries = glob_with(pattern, MatchOptions::new()).unwrap();
90    for entry in entries.filter_map(Result::ok) {
91        let path = entry.as_path().into();
92        generate_struct_single(files, path, cmd)?;
93    }
94    Ok(())
95}

112fn generate_struct_from_ir(ir: &FeatureManifest, cmd: &GenerateStructCmd) -> Result<()> {
113    let language = &cmd.language;
114    ir.validate_manifest_for_lang(language)?;
115    match language {
116        TargetLanguage::IR => {
117            let contents = serde_json::to_string_pretty(&ir)?;
118            std::fs::write(&cmd.output, contents)?;
119        }
120        TargetLanguage::Kotlin => backends::kotlin::generate_struct(ir, cmd)?,
121        TargetLanguage::Swift => backends::swift::generate_struct(ir, cmd)?,
122        _ => unimplemented!(
123            "Unsupported output language for structs: {}",
124            language.extension()
125        ),
126    };
127    Ok(())
128}

162    load_from_ir: bool,

210fn wrapped(text: &str, initial_indent: &str, subsequent_indent: &str) -> String {
211    let options = textwrap::Options::new(HELP_WIDTH)
212        .initial_indent(initial_indent)
213        .subsequent_indent(subsequent_indent)
214        // Lint prose is full of kebab-case names; don't break them at the hyphens.
215        .word_splitter(textwrap::WordSplitter::NoHyphenation);
216    textwrap::fill(text, options)
217}

219fn output_finding(stream: &mut impl WriteColor, finding: &Finding) -> Result<()> {
220    let (icon, color) = match finding.level {
221        LintLevel::Error => ("❎", Color::Red),
222        _ => ("⚠️", Color::Yellow),
223    };
224
225    write!(stream, "  {icon} ")?;
226    stream.set_color(ColorSpec::new().set_fg(Some(color)))?;
227    write!(stream, "{:<LINT_NAME_WIDTH$}", finding.lint)?;
228    stream.reset()?;
229    writeln!(stream, "{}", finding.message)?;
230
231    Ok(())
232}

234/// Print the findings grouped under the feature, object or enum they're about,
235/// followed by the guidance for each lint that fired. The guidance is per lint, not
236/// per finding: eighty features missing a `meta-bug` need telling how once.
237fn output_findings(stream: &mut impl WriteColor, report: &LintReport) -> Result<()> {
238    // The findings arrive sorted by module and subject, so a change of either starts
239    // the next group.
240    let mut group: Option<(&Option<String>, &String)> = None;
241
242    for finding in &report.findings {
243        let this = (&finding.module, &finding.subject);
244        if group != Some(this) {
245            if group.is_some() {
246                writeln!(stream)?;
247            }
248            group = Some(this);
249
250            stream.set_color(ColorSpec::new().set_bold(true))?;
251            write!(stream, "{}", finding.subject)?;
252            stream.reset()?;
253            match &finding.module {
254                Some(module) => writeln!(stream, " (imported from {module})")?,
255                None => writeln!(stream)?,
256            }
257        }
258        output_finding(stream, finding)?;
259    }
260
261    let lints = report.triggered_lints();
262    if !lints.is_empty() {
263        writeln!(stream, "\nWhat to do about these:")?;
264        for lint in lints {
265            stream.set_color(ColorSpec::new().set_bold(true))?;
266            writeln!(stream, "  {}", lint.name)?;
267            stream.reset()?;
268            writeln!(stream, "{}", wrapped(lint.help, "    ", "    "))?;
269        }
270        writeln!(stream)?;
271    }
272
273    Ok(())
274}

286pub(crate) fn validate(cmd: &ValidateCmd) -> Result<()> {
287    let mut stdout = stdout_stream();
288
289    let files: FileLoader = TryFrom::try_from(&cmd.loader)?;
290
291    let filename = &cmd.manifest;
292    let file_path = files.file_path(filename)?;
293    let parser: Parser = Parser::new(files, file_path.clone())?;
294    let mut loading = HashSet::new();
295    let manifest_front_end = parser.load_manifest(&file_path, &mut loading)?;
296
297    let iter_includes = loading.iter().map(|id| id.to_string());
298
299    let channels = manifest_front_end.channels();
300    if channels.is_empty() {
301        output_note(
302            &mut stdout,
303            &format!(
304                "Loaded modules:\n- {}\n",
305                iter_includes.collect::<Vec<String>>().join("\n- ")
306            ),
307        )?;
308        output_ok(&mut stdout, &format!(
309            "{}\n{}\n{}",
310            "The manifest is valid for including in other files. To be imported, or used as an app manifest, it requires the following:",
311            "- A `channels` list",
312            "- An `about` block",
313        ))?;
314        return Ok(());
315    }
316    let intermediate_representation =
317        parser
318            .get_intermediate_representation(None)
319            .inspect_err(|e| {
320                output_err(&mut stdout, "Manifest is invalid", &e.to_string()).unwrap();
321            })?;
322
323    output_note(
324        &mut stdout,
325        &format!(
326            "Loaded modules:\n- {}\n",
327            iter_includes
328                .chain(
329                    intermediate_representation
330                        .all_imports
331                        .keys()
332                        .map(|m| m.to_string())
333                )
334                .collect::<Vec<String>>()
335                .join("\n- ")
336        ),
337    )?;
338
339    writeln!(&mut stdout, "Validating manifest for different channels:")?;
340
341    let results = channels
342        .iter()
343        .map(|c| {
344            let intermediate_representation = parser.get_intermediate_representation(Some(c));
345            match intermediate_representation {
346                Ok(ir) => (
347                    c,
348                    ir.validate_manifest_with(cmd.loader.lax_gecko_pref_validation),
349                ),
350                Err(e) => (c, Err(e)),
351            }
352        })
353        .collect::<Vec<(&String, Result<_>)>>();
354
355    let mut error_count = 0;
356    for (channel, result) in results {
357        match result {
358            Ok(_) => {
359                output_ok(&mut stdout, &format!("{channel:.<20}valid"))?;
360            }
361            Err(e) => {
362                error_count += 1;
363                output_err(
364                    &mut stdout,
365                    &format!("{channel:.<20}invalid"),
366                    &e.to_string(),
367                )?;
368            }
369        };
370    }
371
372    if error_count > 0 {
373        return Err(CliError(format!(
374            "Manifest contains error(s) in {} channel{}",
375            error_count,
376            if error_count > 1 { "s" } else { "" }
377        )));
378    }
379
380    Ok(())
381}

383fn lint_report(cmd: &LintCmd) -> Result<LintReport> {
384    let files: FileLoader = TryFrom::try_from(&cmd.loader)?;
385    let file_path = files.file_path(&cmd.manifest)?;
386
387    // One parser for both passes: `load_manifest` walks the whole include tree, and
388    // a second one would fetch and parse all of it again.
389    let parser: Parser = Parser::new(files, file_path.clone())?;
390
391    // The top level `no-lint` block lives in the file, not the IR.
392    let mut loading = HashSet::new();
393    let manifest_front_end = parser.load_manifest(&file_path, &mut loading)?;
394
395    let config = LintConfig::new()
396        .including_imports(cmd.include_imports)
397        .with_file_suppressions(&manifest_front_end.no_lint)
398        .allowing(&cmd.allow)?
399        .denying(&cmd.deny)?;
400
401    // Linting an invalid manifest would report nonsense.
402    let ir = parser
403        .get_intermediate_representation(None)
404        .and_then(|ir| {
405            ir.validate_manifest_with(cmd.loader.lax_gecko_pref_validation)
406                .map(|_| ir)
407        })
408        .map_err(|e| {
409            CliError(format!(
410                "{e}\nA manifest has to be valid before it can be linted; run `nimbus-fml validate` for details"
411            ))
412        })?;
413
414    Ok(lints::lint_manifest(&ir, &config))
415}

417pub(crate) fn lint(cmd: &LintCmd) -> Result<()> {
418    let mut stdout = stdout_stream();
419
420    let report = lint_report(cmd)?;
421
422    if cmd.as_json {
423        println!("{}", serde_json::to_string_pretty(&json_report(&report))?);
424    } else {
425        output_findings(&mut stdout, &report)?;
426        output_lint_summary(&mut stdout, &report)?;
427    }
428
429    let errors = report.error_count();
430    let warnings = report.warning_count();
431
432    if errors > 0 {
433        return Err(CliError(format!(
434            "Manifest has {} lint error{}",
435            errors,
436            if errors > 1 { "s" } else { "" }
437        )));
438    }
439
440    if cmd.error_on_warning && warnings > 0 {
441        return Err(CliError(format!(
442            "Manifest has {} lint warning{}",
443            warnings,
444            if warnings > 1 { "s" } else { "" }
445        )));
446    }
447
448    Ok(())
449}

451/// The shape `--json` emits: the counts a CI job needs, plus the findings.
452fn json_report(report: &LintReport) -> serde_json::Value {
453    serde_json::json!({
454        "errors": report.error_count(),
455        "warnings": report.warning_count(),
456        "suppressed": report.suppressed,
457        "subjects": report.subject_count(),
458        "findings": report.findings,
459    })
460}

462fn output_suppressed(stream: &mut impl WriteColor, report: &LintReport) -> Result<()> {
463    if report.suppressed == 0 {
464        return Ok(());
465    }
466    output_note(
467        stream,
468        &format!(
469            "{} finding{} silenced by `no-lint`",
470            report.suppressed,
471            if report.suppressed > 1 { "s" } else { "" }
472        ),
473    )
474}

476fn output_lint_summary(stream: &mut impl WriteColor, report: &LintReport) -> Result<()> {
477    if report.is_empty() {
478        output_ok(stream, "No lint findings")?;
479        return output_suppressed(stream, report);
480    }
481
482    let errors = report.error_count();
483    let warnings = report.warning_count();
484    let mut counts = Vec::new();
485    if errors > 0 {
486        counts.push(format!(
487            "{errors} error{}",
488            if errors > 1 { "s" } else { "" }
489        ));
490    }
491    if warnings > 0 {
492        counts.push(format!(
493            "{warnings} warning{}",
494            if warnings > 1 { "s" } else { "" }
495        ));
496    }
497
498    let subjects = report.subject_count();
499    writeln!(
500        stream,
501        "Found {} in {subjects} place{}.",
502        counts.join(" and "),
503        if subjects > 1 { "s" } else { "" }
504    )?;
505    output_suppressed(stream, report)?;
506    writeln!(
507        stream,
508        "{}",
509        wrapped(
510            "A lint that doesn't apply can be switched off for a single feature with a `no-lint: [LINT_NAME]` list on that feature, for the whole file with a top level `no-lint:` list, or for this run with `--allow LINT_NAME`.",
511            "",
512            "",
513        )
514    )?;
515
516    Ok(())
517}

538            lint.default_level.as_str(),
539            lint.description

541    }

686        test_script: &str,
687        manifests: &[(&str, &str)],
688    ) -> Result<()> {
689        let cmds = manifests
690            .iter()
691            .map(|(manifest, channel)| {

706            .iter()
707            .map(|cmd| cmd.output.display().to_string())
708            .collect::<Vec<_>>();
709
710        run_script_with_generated_code(language, &manifests_out, test_script)?;
711        Ok(())
712    }

1304mod test_jsonschema {
1305    use std::fs;
1306
1307    use jsonschema::JSONSchema;
1308    use tempfile::NamedTempFile;
1309
1310    use super::test::{create_experimenter_manifest_cmd, MANIFEST_PATHS};
1311    use super::*;
1312    use crate::backends::experimenter_manifest::ExperimenterManifest;
1313    use crate::util::{join, pkg_dir};
1314
1315    fn validate_against_experimenter_schema<P: AsRef<Path>>(
1316        schema_path: P,
1317        generated_yaml: &serde_yaml::Value,
1318    ) -> Result<()> {
1319        let generated_manifest: ExperimenterManifest =
1320            serde_yaml::from_value(generated_yaml.to_owned())?;
1321        let generated_json = serde_json::to_value(generated_manifest)?;
1322
1323        let schema = fs::read_to_string(&schema_path)?;
1324        let schema: serde_json::Value = serde_json::from_str(&schema)?;
1325        let compiled = JSONSchema::compile(&schema).expect("The schema is invalid");
1326        let res = compiled.validate(&generated_json);
1327        if let Err(e) = res {
1328            panic!(
1329                "Validation errors: \n{}",
1330                e.map(|e| e.to_string()).collect::<Vec<String>>().join("\n")
1331            );
1332        }
1333        Ok(())
1334    }
1335
1336    #[test]
1337    fn test_schema_validation() -> Result<()> {
1338        for path in MANIFEST_PATHS {
1339            let output = NamedTempFile::new()?;
1340            let cmd = create_experimenter_manifest_cmd(path, output.path())?;
1341            generate_experimenter_manifest(&cmd)?;
1342
1343            let generated = fs::read_to_string(&cmd.output)?;
1344            let generated_yaml = serde_yaml::from_str(&generated)?;
1345            validate_against_experimenter_schema(
1346                join(pkg_dir(), "ExperimentFeatureManifest.schema.json"),
1347                &generated_yaml,
1348            )?;
1349        }
1350        Ok(())
1351    }
