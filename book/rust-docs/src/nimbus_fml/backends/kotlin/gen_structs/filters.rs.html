68#[askama::filter_fn]
69pub fn preference_getter<T, P, K>(
70    type_: T,
71    _: &dyn askama::Values,
72    prefs: P,
73    pref_key: K,
74) -> Result<String, askama::Error>
75where
76    T: Borrow<TypeIdentifier>,
77    P: fmt::Display,
78    K: fmt::Display,
79{
80    let oracle = &ConcreteCodeOracle;
81    let ct = oracle.find(type_.borrow());
82    if let Some(getter) = ct.preference_getter(oracle, &prefs, &pref_key) {
83        Ok(getter)
84    } else {
85        unreachable!("The preference for type {} isn't available. This is a bug in Nimbus FML Kotlin generator", type_.borrow());
86    }
87}

127#[askama::filter_fn]
128pub fn comment<T>(txt: T, _: &dyn askama::Values, spaces: &str) -> Result<String, askama::Error>
129where
130    T: fmt::Display,
131{
132    use textwrap::{fill, Options};
133
134    let indent_start = "/** ".to_string();
135    let indent_mid = format!("{} * ", spaces);
136    let indent_end = format!("{} */", spaces);
137
138    let options = Options::new(80)
139        .initial_indent(&indent_mid)
140        .subsequent_indent(&indent_mid);
141
142    let lines = fill(txt.to_string().as_str(), options);
143    Ok(format!(
144        "{start}\n{lines}\n{indent}",
145        start = indent_start,
146        lines = lines,
147        indent = indent_end
148    ))
149}
