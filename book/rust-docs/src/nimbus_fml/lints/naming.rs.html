19define_lints! {
20    FEATURE_NAME_CASING: Naming, Warning =
21        "Feature ids should be kebab-case.",
22        "Feature ids are typed by hand into Experimenter.";
23    VARIABLE_NAME_CASING: Naming, Warning =
24        "Variable and field names should be kebab-case.",
25        "Variables are written out as JSON keys in Experimenter.";
26    TYPE_NAME_CASING: Naming, Warning =
27        "Objects and enums should be UpperCamelCase.",
28        "Objects and enums become classes and types in the generated Kotlin and Swift.";
29    ENUM_VARIANT_CASING: Naming, Warning =
30        "Enum variants should be kebab-case.",
31        "Variants are written out as JSON strings in Experimenter.";
32    COMMON_PREFIX: Naming, Warning =
33        "Variables shouldn't repeat the name of the feature they belong to.",
34        "Variables are always read together with the feature they belong to, so a shared prefix only makes them longer. Drop it, or group the variables into an object if the prefix is really a thing in its own right.";
35    TYPE_IN_NAME: Naming, Warning =
36        "Variable names shouldn't repeat the name of their type.",
37        "The type is shown next to the name everywhere the name is; repeating it only makes the name longer.";
38    NEGATED_BOOLEAN: Naming, Warning =
39        "Booleans should be named for what is true, not what is false.",
40        "Everyone setting this in an experiment has to work out what `false` means. Name the variable for the behaviour that is switched on, and flip the default.";
41}

43lazy_static! {
44    static ref KEBAB_CASE: Regex = Regex::new(r"^[a-z][a-z0-9]*(-[a-z0-9]+)*$").unwrap();
45    static ref UPPER_CAMEL_CASE: Regex = Regex::new(r"^[A-Z][A-Za-z0-9]*$").unwrap();
46}

48/// Words that read as a predicate rather than a namespace, so sharing one is a
49/// convention rather than a prefix to strip.
50const PREDICATE_WORDS: &[&str] = &["allow", "can", "has", "is", "should", "show", "use"];
51
52/// Words that make `false` mean the thing happens.
53const NEGATIVE_WORDS: &[&str] = &[
54    "disable",
55    "disabled",
56    "disallow",
57    "disallowed",
58    "dont",
59    "hidden",
60    "hide",
61    "never",
62    "no",
63    "not",
64    "prevent",
65    "suppress",
66];

93fn check_feature(feature: &FeatureDef, out: &mut Vec<RawFinding>) {
94    if !KEBAB_CASE.is_match(&feature.name) {
95        out.push(RawFinding::new(
96            &FEATURE_NAME_CASING,
97            feature_path(feature),
98            format!(
99                "`{}` isn't kebab-case{}",
100                feature.name,
101                rename_to(&feature.name)
102            ),
103        ));
104    }
105
106    for prop in &feature.props {
107        check_variable_name(&prop.name, variable_path(feature, prop), out);
108        check_type_in_name(prop, variable_path(feature, prop), out);
109        check_negated_boolean(prop, variable_path(feature, prop), out);
110    }
111
112    check_common_prefix(feature, out);
113}
114
115fn check_object(object: &ObjectDef, out: &mut Vec<RawFinding>) {
116    if !UPPER_CAMEL_CASE.is_match(&object.name) {
117        out.push(RawFinding::new(
118            &TYPE_NAME_CASING,
119            object_path(object),
120            format!("`{}` isn't UpperCamelCase", object.name),
121        ));
122    }
123
124    for prop in &object.props {
125        check_variable_name(&prop.name, object_field_path(object, prop), out);
126        check_type_in_name(prop, object_field_path(object, prop), out);
127        check_negated_boolean(prop, object_field_path(object, prop), out);
128    }
129}
130
131fn check_enum(enum_def: &EnumDef, out: &mut Vec<RawFinding>) {
132    if !UPPER_CAMEL_CASE.is_match(&enum_def.name) {
133        out.push(RawFinding::new(
134            &TYPE_NAME_CASING,
135            enum_path(enum_def),
136            format!("`{}` isn't UpperCamelCase", enum_def.name),
137        ));
138    }
139
140    for variant in &enum_def.variants {
141        if !KEBAB_CASE.is_match(&variant.name) {
142            out.push(RawFinding::new(
143                &ENUM_VARIANT_CASING,
144                enum_variant_path(enum_def, &variant.name),
145                format!(
146                    "`{}` isn't kebab-case{}",
147                    variant.name,
148                    rename_to(&variant.name)
149                ),
150            ));
151        }
152    }
153}
154
155fn check_variable_name(name: &str, path: Location, out: &mut Vec<RawFinding>) {
156    if !KEBAB_CASE.is_match(name) {
157        out.push(RawFinding::new(
158            &VARIABLE_NAME_CASING,
159            path,
160            format!("`{name}` isn't kebab-case{}", rename_to(name)),
161        ));
162    }
163}

165/// `sections-list: List<Section>` says "list" twice.
166fn check_type_in_name(prop: &PropDef, path: Location, out: &mut Vec<RawFinding>) {
167    let Some((_, last)) = prop.name.rsplit_once('-') else {
168        return;
169    };
170    let last = last.to_ascii_lowercase();
171
172    // `Option<Boolean>` is still a boolean, as far as the name goes.
173    let typ = match &prop.typ {
174        TypeRef::Option(inner) => inner.as_ref(),
175        typ => typ,
176    };
177
178    let redundant = match typ {
179        TypeRef::Boolean => ["bool", "boolean", "flag"].contains(&last.as_str()),
180        TypeRef::Int => ["int", "integer", "num", "number"].contains(&last.as_str()),
181        TypeRef::String => ["str", "string"].contains(&last.as_str()),
182        TypeRef::List(_) => ["list", "array"].contains(&last.as_str()),
183        TypeRef::StringMap(_) | TypeRef::EnumMap(..) => {
184            ["map", "dict", "dictionary"].contains(&last.as_str())
185        }
186        TypeRef::Enum(_) => last == "enum",
187        TypeRef::Object(_) => ["obj", "object", "json"].contains(&last.as_str()),
188        _ => false,
189    };
190
191    if redundant {
192        out.push(RawFinding::new(
193            &TYPE_IN_NAME,
194            path,
195            format!(
196                "`{}` ends in `-{last}`, but its type is already `{}`",
197                prop.name, prop.typ
198            ),
199        ));
200    }
201}

203/// `disable-sync: false` takes a moment to read; `sync-enabled: true` doesn't.
204fn check_negated_boolean(prop: &PropDef, path: Location, out: &mut Vec<RawFinding>) {
205    if !is_boolean(&prop.typ) {
206        return;
207    }
208
209    let negative = prop
210        .name
211        .split('-')
212        .find(|word| NEGATIVE_WORDS.contains(&word.to_ascii_lowercase().as_str()));
213
214    if let Some(word) = negative {
215        out.push(RawFinding::new(
216            &NEGATED_BOOLEAN,
217            path,
218            format!("`{}` is a boolean named with `{word}`", prop.name),
219        ));
220    }
221}

223/// A feature called `homescreen` doesn't need variables called `homescreen-*`.
224fn check_common_prefix(feature: &FeatureDef, out: &mut Vec<RawFinding>) {
225    let feature_prefix = format!("{}-", feature.name);
226    let mut prefixed = Vec::new();
227
228    for prop in &feature.props {
229        if prop.name.starts_with(&feature_prefix) {
230            prefixed.push(prop);
231        }
232    }
233
234    for prop in &prefixed {
235        out.push(RawFinding::new(
236            &COMMON_PREFIX,
237            variable_path(feature, prop),
238            format!(
239                "`{}` repeats the name of the feature it belongs to; rename it to `{}`",
240                prop.name,
241                &prop.name[feature_prefix.len()..]
242            ),
243        ));
244    }
245
246    // A prefix shared by every variable is noise even when it isn't the feature name.
247    if prefixed.is_empty() && feature.props.len() > 1 {
248        if let Some(shared) = shared_prefix(&feature.props) {
249            out.push(RawFinding::new(
250                &COMMON_PREFIX,
251                feature_path(feature),
252                format!(
253                    "All {} variables of this feature start with `{shared}-`",
254                    feature.props.len()
255                ),
256            ));
257        }
258    }
259}
260
261fn shared_prefix(props: &[PropDef]) -> Option<String> {
262    let first = props.first()?.name.split('-').next()?.to_string();
263    if PREDICATE_WORDS.contains(&first.as_str()) {
264        return None;
265    }
266    props
267        .iter()
268        .all(|p| {
269            p.name
270                .split_once('-')
271                .map(|(head, _)| head == first)
272                .unwrap_or_default()
273        })
274        .then_some(first)
275}

277/// Empty when `heck` can't improve on `name`, as for `9lives`.
278fn rename_to(name: &str) -> String {
279    use heck::ToKebabCase;
280    let kebab = name.to_kebab_case();
281    if kebab == name {
282        String::new()
283    } else {
284        format!("; rename it to `{kebab}`")
285    }
286}

293    fn prop(name: &str, typ: &TypeRef) -> PropDef {
294        PropDef::with_doc(name, "A description of the variable.", typ, &json!(null))

297    fn feature(name: &str, props: Vec<PropDef>) -> FeatureDef {
298        FeatureDef::new(name, "A description of the feature.", props, false)

301    fn lints(feature: &FeatureDef) -> Vec<&'static str> {
302        let mut out = Vec::new();
303        check_feature(feature, &mut out);
304        out.iter().map(|f| f.lint.name).collect()
305    }
