17define_lints! {
18    NO_VARIABLES: Design =
19        "Features should have something an experiment can change.",
20        "An experiment on a feature with no variables can only enrol users; it can't change what they see. Add at least a boolean `enabled` variable.";
21    TOO_MANY_VARIABLES: Design =
22        "Features with a lot of variables are hard to configure correctly.",
23        "Every variable is something an experiment owner has to understand. Consider splitting the feature up, or grouping related variables into objects.";
24    DEEP_NESTING: Design =
25        "Deeply nested configuration is hard to write by hand.",
26        "Experiment owners write these values out as JSON by hand. Consider flattening the value, or adding an `examples` block showing a complete one.";
27    TRIVIAL_ENUM: Design =
28        "An enum should offer a choice.",
29        "An enum that can only take one value can't be varied by an experiment. Add the other variants, or use a boolean.";
30    UNUSED_TYPE: Design =
31        "Objects and enums should be used by at least one feature.",
32        "Unused types are still generated into Kotlin and Swift, and still have to be maintained. Delete it, or use it.";
33}
34
35const MAX_VARIABLES: usize = 25;
36
37/// How many nested levels of objects are allowed.
38///
39/// A feature with no object types has a depth of 1.
40const MAX_NESTING_DEPTH: usize = 3;
41
42/// Lints about the design of features.
43pub(crate) struct DesignLints;
44
45impl Linter for DesignLints {
46    fn lints(&self) -> &'static [&'static LintInfo] {
47        LINTS
48    }
49
50    fn lint_feature(
51        &self,
52        feature: &FeatureDef,
53        manifest: &FeatureManifest,
54        out: &mut Vec<UnreportedIssue>,
55    ) {
56        lint_feature(feature, manifest, out);
57    }
58
59    fn lint_enum(&self, enum_def: &EnumDef, out: &mut Vec<UnreportedIssue>) {
60        lint_enum(enum_def, out);
61    }
62
63    fn lint_manifest(&self, manifest: &FeatureManifest, out: &mut Vec<UnreportedIssue>) {
64        lint_manifest(manifest, out);
65    }

68fn lint_feature(feature: &FeatureDef, manifest: &FeatureManifest, out: &mut Vec<UnreportedIssue>) {
69    if feature.props.is_empty() {
70        out.push(UnreportedIssue::new(
71            &NO_VARIABLES,
72            feature_path(feature),
73            "This feature has no variables",
74        ));
75        // The rest is about variables this feature doesn't have.
76        return;
77    }
78
79    if feature.props.len() > MAX_VARIABLES {
80        out.push(UnreportedIssue::new(
81            &TOO_MANY_VARIABLES,
82            feature_path(feature),
83            format!(
84                "This feature has {} variables (the limit is {MAX_VARIABLES})",
85                feature.props.len()
86            ),
87        ));
88    }
89
90    for prop in &feature.props {
91        lint_feature_nesting(feature, prop, &manifest.obj_defs, out);
92    }
93}
94
95fn lint_enum(enum_def: &EnumDef, out: &mut Vec<UnreportedIssue>) {
96    if enum_def.variants.len() < 2 {
97        out.push(UnreportedIssue::new(
98            &TRIVIAL_ENUM,
99            enum_path(enum_def),
100            format!(
101                "This enum has {} variant{}",
102                enum_def.variants.len(),
103                if enum_def.variants.len() == 1 {
104                    ""
105                } else {
106                    "s"
107                }
108            ),
109        ));
110    }
111}
112
113fn lint_manifest(manifest: &FeatureManifest, out: &mut Vec<UnreportedIssue>) {
114    if manifest.feature_defs.is_empty() {
115        return;
116    }
117
118    let mut used = HashSet::new();
119    for feature in manifest.iter_feature_defs() {
120        for prop in &feature.props {
121            mark_used(&prop.typ, &manifest.obj_defs, &mut used);
122        }
123    }
124
125    for object in manifest.iter_object_defs() {
126        if !used.contains(&TypeRef::Object(object.name.clone())) {
127            out.push(UnreportedIssue::new(
128                &UNUSED_TYPE,
129                object_path(object),
130                "No feature in this manifest uses this object",
131            ));
132        }
133    }
134
135    for enum_def in manifest.iter_enum_defs() {
136        if !used.contains(&TypeRef::Enum(enum_def.name.clone())) {
137            out.push(UnreportedIssue::new(
138                &UNUSED_TYPE,
139                enum_path(enum_def),
140                "No feature in this manifest uses this enum",
141            ));
142        }
143    }
144}
145
146/// Record every type reachable from `typ`.
147fn mark_used(typ: &TypeRef, objects: &BTreeMap<String, ObjectDef>, used: &mut HashSet<TypeRef>) {
148    if !used.insert(typ.clone()) {
149        return;
150    }
151
152    match typ {
153        TypeRef::Option(inner) | TypeRef::List(inner) | TypeRef::StringMap(inner) => {
154            mark_used(inner, objects, used)
155        }
156        TypeRef::EnumMap(keys, values) => {
157            mark_used(keys, objects, used);
158            mark_used(values, objects, used);
159        }
160        TypeRef::Object(name) => {
161            if let Some(object) = objects.get(name) {
162                for prop in &object.props {
163                    mark_used(&prop.typ, objects, used);
164                }
165            }
166        }
167        _ => {}
168    }
169}

171fn lint_feature_nesting(
172    feature: &FeatureDef,
173    prop: &PropDef,
174    objects: &BTreeMap<String, ObjectDef>,
175    out: &mut Vec<UnreportedIssue>,
176) {
177    let mut visiting = HashSet::new();
178    let depth = 1 + compute_type_depth(&prop.typ, objects, &mut visiting);
179    if depth > MAX_NESTING_DEPTH {
180        out.push(UnreportedIssue::new(
181            &DEEP_NESTING,
182            variable_path(feature, prop),
183            format!(
184                "The value of `{}` is {depth} levels deep (the limit is {MAX_NESTING_DEPTH})",
185                prop.name
186            ),
187        ));
188    }
189}
190
191/// How many levels of objects are underneath this type.
192fn compute_type_depth(
193    typ: &TypeRef,
194    objects: &BTreeMap<String, ObjectDef>,
195    visiting: &mut HashSet<String>,
196) -> usize {
197    match typ {
198        TypeRef::Option(inner) | TypeRef::List(inner) | TypeRef::StringMap(inner) => {
199            compute_type_depth(inner, objects, visiting)
200        }
201        TypeRef::EnumMap(_, values) => compute_type_depth(values, objects, visiting),
202        TypeRef::Object(name) => {
203            if !visiting.insert(name.clone()) {
204                // Stop at a cycle; it is already deep enough to report.
205                return MAX_NESTING_DEPTH;
206            }
207            let depth = objects
208                .get(name)
209                .map(|o| {
210                    o.props
211                        .iter()
212                        .map(|p| compute_type_depth(&p.typ, objects, visiting))
213                        .max()
214                        .unwrap_or_default()
215                })
216                .unwrap_or_default();
217            visiting.remove(name);
218            1 + depth
219        }
220        _ => 0,
221    }
222}

241    fn lints_with(feature: &FeatureDef, manifest: &FeatureManifest) -> Vec<&'static str> {

261    #[test]
262    fn test_deep_nesting() {
263        let mut manifest = FeatureManifest {
264            obj_defs: ObjectDef::into_map(&[
265                ObjectDef::new(
266                    "Outer",
267                    &[prop("middle", &TypeRef::Object("Middle".into()))],
268                ),
269                ObjectDef::new("Middle", &[prop("inner", &TypeRef::Object("Inner".into()))]),
270                ObjectDef::new("Inner", &[prop("label", &TypeRef::String)]),
271            ]),
272            ..Default::default()
273        };
274
275        // feature › Outer › Middle › Inner is 4 levels.
276        let outer = feature(vec![prop("outer", &TypeRef::Object("Outer".into()))]);
277        assert!(lints_with(&outer, &manifest).contains(&"DEEP_NESTING"));
278
279        // feature › Middle › Inner is 3.
280        let middle = feature(vec![prop("middle", &TypeRef::Object("Middle".into()))]);
281        assert!(!lints_with(&middle, &manifest).contains(&"DEEP_NESTING"));
282
283        // A list of objects is as deep as the objects in it.
284        let list_of_outers = feature(vec![prop(
285            "outers",
286            &TypeRef::List(Box::new(TypeRef::Object("Outer".into()))),
287        )]);
288        assert!(lints_with(&list_of_outers, &manifest).contains(&"DEEP_NESTING"));
289
290        // A cycle terminates.
291        manifest.obj_defs.insert(
292            "Inner".to_string(),
293            ObjectDef::new("Inner", &[prop("outer", &TypeRef::Object("Outer".into()))]),
294        );
295        assert!(lints_with(&outer, &manifest).contains(&"DEEP_NESTING"));
296    }
