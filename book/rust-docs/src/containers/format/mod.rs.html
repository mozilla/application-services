16/// Reads a stored document, applying every migration needed to reach
17/// [`LATEST_VERSION`]. The flag reports whether a migration ran, and the
18/// document therefore has to be written back even though nothing the user did
19/// changed it.
20///
21/// An error means the stored data is unusable and the store has to be seeded
22/// from the defaults, discarding the data held by the previous containers.
23pub(crate) fn parse(bytes: &[u8]) -> Result<(ContainersData, bool), ParseError> {
24    let mut data: ContainersData = serde_json::from_slice(bytes)?;
25
26    // Version 1 predates every migration path.
27    if data.version == 1 {
28        return Err(ParseError::UnsupportedVersion(1));
29    }
30
31    let mut migrated = false;
32
33    if data.version == 2 {
34        migrations::migrate_2_to_3(&mut data);
35        migrated = true;
36    }
37
38    if data.version == 3 {
39        migrations::migrate_3_to_4(&mut data);
40        migrated = true;
41    }
42
43    if data.version == 4 {
44        migrations::migrate_4_to_5(&mut data);
45        migrated = true;
46    }
47
48    if data.version == 5 {
49        migrations::migrate_5_to_6(&mut data);
50        migrated = true;
51    }
52
53    if data.version == 6 {
54        migrations::migrate_6_to_7(&mut data);
55        migrated = true;
56    }
57
58    if data.version == 7 {
59        migrations::migrate_7_to_8(&mut data);
60        migrated = true;
61    }
62
63    if data.version != LATEST_VERSION {
64        return Err(ParseError::UnsupportedVersion(data.version));
65    }
66
67    Ok((data, migrated))
68}

71    serde_json::to_vec(data).expect("containers data is always serializable")
