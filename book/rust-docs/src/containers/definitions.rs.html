7/// Everything a color is, kept on one line per color. Borrowed, so it stays
8/// inside the crate: borrowed data cannot cross an FFI boundary.
9#[derive(Clone, Copy)]
10struct ColorDef {
11    name: &'static str,
12    code: &'static str,
13    code_nova: &'static str,
14    gecko_l10n_id: &'static str,
15}
16
17/// The colors a container can carry.
18///
19/// A stored document holds the name, not the variant, and the same caveats as
20/// [`ContainerIcon`] apply. Legacy names are not variants: they are resolved to
21/// their canonical replacement by [`resolve_color`].
22#[derive(Clone, Copy, Debug, PartialEq, Eq, uniffi::Enum)]
23pub enum ContainerColor {
24    Gray,
25    Yellow,
26    Orange,
27    Red,
28    Pink,
29    Purple,
30    Violet,
31    Blue,
32    Cyan,
33    Green,
34}
35
36impl ContainerColor {
37    /// Every color, in the order the container editor offers them.
38    pub const ALL: &'static [Self] = &[
39        Self::Gray,
40        Self::Yellow,
41        Self::Orange,
42        Self::Red,
43        Self::Pink,
44        Self::Purple,
45        Self::Violet,
46        Self::Blue,
47        Self::Cyan,
48        Self::Green,
49    ];
50
51    fn def(self) -> ColorDef {
52        /// Positional, in the order [`ColorDef`] declares: the Fluent id is
53        /// spelled out rather than built from the name, so that Firefox can
54        /// rename one without renaming the other.
55        macro_rules! def {
56            ($name:literal, $code:literal, $code_nova:literal, $gecko_l10n_id:literal) => {
57                ColorDef {
58                    name: $name,
59                    code: $code,
60                    code_nova: $code_nova,
61                    gecko_l10n_id: $gecko_l10n_id,
62                }
63            };
64        }
65
66        match self {
67            Self::Gray => def!("gray", "#7c7c7d", "#949297", "user-context-color-gray"),
68            Self::Yellow => def!("yellow", "#ffcb00", "#db820e", "user-context-color-yellow"),
69            Self::Orange => def!("orange", "#ff9f00", "#f4682c", "user-context-color-orange"),
70            Self::Red => def!("red", "#ff613d", "#ed566e", "user-context-color-red"),
71            Self::Pink => def!("pink", "#ff4bda", "#db54bf", "user-context-color-pink"),
72            Self::Purple => def!("purple", "#af51f5", "#b864ee", "user-context-color-purple"),
73            Self::Violet => def!("violet", "#764edd", "#9871ff", "user-context-color-violet"),
74            Self::Blue => def!("blue", "#37adff", "#5a87fd", "user-context-color-blue"),
75            Self::Cyan => def!("cyan", "#00c79a", "#10a4ca", "user-context-color-cyan"),
76            Self::Green => def!("green", "#51cd00", "#11ae84", "user-context-color-green"),
77        }
78    }
79
80    /// The name the document stores.
81    pub fn name(self) -> &'static str {
82        self.def().name
83    }
84
85    /// The canonical color a stored document, an enterprise policy or a
86    /// WebExtension names, or `None` when the crate has no such color. A legacy
87    /// name has to go through [`resolve_color`] first.
88    pub fn from_name(name: &str) -> Option<Self> {
89        Self::ALL.iter().copied().find(|color| color.name() == name)
90    }
91
92    /// `nova` picks the refreshed value over the legacy one. Which to use
93    /// depends on a setting the embedder owns, so the caller decides.
94    pub fn code(self, nova: bool) -> &'static str {
95        let def = self.def();
96        if nova {
97            def.code_nova
98        } else {
99            def.code
100        }
101    }
102
103    /// The Fluent id Firefox Desktop labels the color with. See
104    /// [`ContainerIcon::gecko_l10n_id`] on why this is per platform.
105    pub fn gecko_l10n_id(self) -> &'static str {
106        self.def().gecko_l10n_id
107    }
108}

110/// How a container gets its label.
111#[derive(Clone, Debug, PartialEq, Eq, uniffi::Enum)]
112pub enum ContainerLabel {
113    Name { name: String },
114    Personal,
115    Work,
116    Banking,
117    Shopping,
118}
119
120impl ContainerLabel {
121    /// The Fluent id Firefox Desktop labels the container with, or `None` for a
122    /// container that carries its own name. See
123    /// [`ContainerIcon::gecko_l10n_id`] on why this is per platform.
124    pub fn gecko_l10n_id(&self) -> Option<&'static str> {
125        match self {
126            Self::Name { .. } => None,
127            Self::Personal => Some("user-context-personal2"),
128            Self::Work => Some("user-context-work2"),
129            Self::Banking => Some("user-context-banking2"),
130            Self::Shopping => Some("user-context-shopping2"),
131        }
132    }
133}

135/// The icons a container can carry. The set is fixed, so a container cannot end
136/// up with an icon the embedder has no artwork for.
137///
138/// A stored document holds the name, not the variant: one written by a newer
139/// Firefox can carry an icon this crate does not know, which reads back as
140/// `None` and is left in the document untouched.
141#[derive(Clone, Copy, Debug, PartialEq, Eq, uniffi::Enum)]
142pub enum ContainerIcon {
143    Fingerprint,
144    Briefcase,
145    Dollar,
146    Cart,
147    Vacation,
148    Gift,
149    Food,
150    Fruit,
151    Pet,
152    Tree,
153    Chill,
154    Circle,
155    Fence,
156}
157
158impl ContainerIcon {
159    /// Every icon, in the order the container editor offers them. The first is
160    /// the one a new container starts with.
161    pub const ALL: &'static [Self] = &[
162        Self::Fingerprint,
163        Self::Briefcase,
164        Self::Dollar,
165        Self::Cart,
166        Self::Vacation,
167        Self::Gift,
168        Self::Food,
169        Self::Fruit,
170        Self::Pet,
171        Self::Tree,
172        Self::Chill,
173        Self::Circle,
174        Self::Fence,
175    ];
176
177    /// The name the document stores.
178    pub fn name(self) -> &'static str {
179        match self {
180            Self::Fingerprint => "fingerprint",
181            Self::Briefcase => "briefcase",
182            Self::Dollar => "dollar",
183            Self::Cart => "cart",
184            Self::Vacation => "vacation",
185            Self::Gift => "gift",
186            Self::Food => "food",
187            Self::Fruit => "fruit",
188            Self::Pet => "pet",
189            Self::Tree => "tree",
190            Self::Chill => "chill",
191            Self::Circle => "circle",
192            Self::Fence => "fence",
193        }
194    }
195
196    /// The icon a stored document, an enterprise policy or a WebExtension
197    /// names, or `None` when the crate has no such icon.
198    pub fn from_name(name: &str) -> Option<Self> {
199        Self::ALL.iter().copied().find(|icon| icon.name() == name)
200    }
201
202    /// The Fluent id Firefox Desktop labels the icon with, so that the embedder
203    /// does not keep a table of its own. Naming schemes do not carry across
204    /// platforms, so each one that needs it gets its own accessor.
205    pub fn gecko_l10n_id(self) -> &'static str {
206        match self {
207            Self::Fingerprint => "user-context-icon-fingerprint",
208            Self::Briefcase => "user-context-icon-briefcase",
209            Self::Dollar => "user-context-icon-dollar",
210            Self::Cart => "user-context-icon-cart",
211            Self::Vacation => "user-context-icon-vacation",
212            Self::Gift => "user-context-icon-gift",
213            Self::Food => "user-context-icon-food",
214            Self::Fruit => "user-context-icon-fruit",
215            Self::Pet => "user-context-icon-pet",
216            Self::Tree => "user-context-icon-tree",
217            Self::Chill => "user-context-icon-chill",
218            Self::Circle => "user-context-icon-circle",
219            Self::Fence => "user-context-icon-fence",
220        }
221    }

224/// Legacy color names, accepted at the WebExtension API boundary and rewritten
225/// to their canonical replacement by the 5 -> 6 migration.
226const ALIASES: &[(&str, &str)] = &[("turquoise", "cyan"), ("toolbar", "gray")];
227
228#[uniffi::export]
229pub fn container_colors() -> Vec<ContainerColor> {
230    ContainerColor::ALL.to_vec()
231}
232
233#[uniffi::export]
234pub fn container_color_aliases() -> HashMap<String, String> {
235    ALIASES
236        .iter()
237        .map(|(legacy, canonical)| (legacy.to_string(), canonical.to_string()))
238        .collect()
239}
240
241#[uniffi::export]
242pub fn resolve_color(name: &str) -> String {
243    ALIASES
244        .iter()
245        .find(|(alias, _)| *alias == name)
246        .map(|(_, canonical)| (*canonical).to_string())
247        .unwrap_or_else(|| name.to_string())
248}

295#[cfg(test)]
296mod tests {
297    use std::collections::HashSet;
298
299    use super::*;
300
301    #[test]
302    fn every_icon_is_named_and_labelled_on_its_own() {
303        let names: HashSet<_> = ContainerIcon::ALL.iter().map(|icon| icon.name()).collect();
304        let l10n_ids: HashSet<_> = ContainerIcon::ALL
305            .iter()
306            .map(|icon| icon.gecko_l10n_id())
307            .collect();
308
309        assert_eq!(names.len(), ContainerIcon::ALL.len());
310        assert_eq!(l10n_ids.len(), ContainerIcon::ALL.len());
311        assert!(ContainerIcon::ALL
312            .iter()
313            .all(|icon| ContainerIcon::from_name(icon.name()) == Some(*icon)));
314    }
315
316    #[test]
317    fn every_color_is_named_and_labelled_on_its_own() {
318        let all = ContainerColor::ALL;
319        let names: HashSet<_> = all.iter().map(|color| color.name()).collect();
320        let l10n_ids: HashSet<_> = all.iter().map(|color| color.gecko_l10n_id()).collect();
321        let codes: HashSet<_> = all
322            .iter()
323            .flat_map(|color| [color.code(false), color.code(true)])
324            .collect();
325
326        assert_eq!(names.len(), all.len());
327        assert_eq!(l10n_ids.len(), all.len());
328        assert_eq!(codes.len(), all.len() * 2, "legacy and refreshed differ");
329        assert!(all
330            .iter()
331            .all(|color| ContainerColor::from_name(color.name()) == Some(*color)));
332    }
333
334    /// A legacy name is not a variant: it resolves to its replacement, and a
335    /// name that is neither is left alone for the document to keep.
336    #[test]
337    fn a_legacy_color_resolves_to_its_replacement() {
338        assert_eq!(resolve_color("turquoise"), "cyan");
339        assert_eq!(resolve_color("toolbar"), "gray");
340        assert_eq!(resolve_color("blue"), "blue");
341        assert_eq!(resolve_color("chartreuse"), "chartreuse");
342
343        assert_eq!(ContainerColor::from_name("turquoise"), None);
344        assert_eq!(
345            ContainerColor::from_name(&resolve_color("turquoise")),
346            Some(ContainerColor::Cyan)
347        );
348    }
349
350    #[test]
351    fn every_default_label_has_its_own_fluent_id() {
352        let defaults = [
353            ContainerLabel::Personal,
354            ContainerLabel::Work,
355            ContainerLabel::Banking,
356            ContainerLabel::Shopping,
357        ];
358        let l10n_ids: HashSet<_> = defaults
359            .iter()
360            .map(|label| label.gecko_l10n_id().expect("a default is localized"))
361            .collect();
362
363        assert_eq!(l10n_ids.len(), defaults.len());
364        assert_eq!(
365            ContainerLabel::Name {
366                name: "Mine".into()
367            }
368            .gecko_l10n_id(),
369            None,
370            "a name the user chose is not localized"
371        );
372    }
