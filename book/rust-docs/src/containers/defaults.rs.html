14/// One of the default containers the store was opened with, shipped or supplied
15/// by the embedder. Enterprise policy can replace the shipped set, and names its
16/// containers rather than localizing them, so its entries carry a
17/// [`ContainerLabel::Name`].
18///
19/// They seed a fresh store, and they label the default containers the user has
20/// not renamed: that label is never stored, so that renaming the string behind
21/// it does not need a migration.
22#[derive(Clone, Debug, PartialEq, Eq, uniffi::Record)]
23pub struct DefaultIdentity {
24    pub icon: ContainerIcon,
25    pub color: ContainerColor,
26    pub label: ContainerLabel,
27}
28
29impl DefaultIdentity {
30    fn new(icon: ContainerIcon, color: ContainerColor, label: ContainerLabel) -> Self {
31        Self { icon, color, label }
32    }
33}

60/// The label of the default identity that owns `user_context_id`, or `None`
61/// when no default does. The nth default owns the nth id, as [`defaults_with`]
62/// hands them out.
63pub(crate) fn default_label(
64    defaults: &[DefaultIdentity],
65    user_context_id: u32,
66) -> Option<&ContainerLabel> {
67    let index = usize::try_from(user_context_id.checked_sub(1)?).ok()?;
68    defaults.get(index).map(|default| &default.label)
69}

71fn system_identity(user_context_id: u32, name: &str) -> Identity {
72    Identity {
73        user_context_id,
74        public: false,
75        icon: String::new(),
76        color: String::new(),
77        name: Some(name.to_string()),
78        policy: false,
79        policy_id: None,
80        extra: Map::new(),
81    }
82}

84pub(crate) fn thumbnail_identity(user_context_id: u32) -> Identity {
85    system_identity(user_context_id, THUMBNAIL_IDENTITY_NAME)
86}

88pub(crate) fn webext_storage_local_identity() -> Identity {

92#[cfg(test)]
93pub(crate) fn defaults() -> ContainersData {
94    defaults_with(&shipped_defaults())
95}

97pub(crate) fn defaults_with(defaults: &[DefaultIdentity]) -> ContainersData {
98    let mut identities = Vec::with_capacity(defaults.len() + 2);
99    let mut next_user_context_id = 1;
100
101    for default in defaults {
102        let name = match &default.label {
103            ContainerLabel::Name { name } => Some(name.clone()),
104            _ => None,
105        };
106
107        identities.push(Identity {
108            user_context_id: next_user_context_id,
109            public: true,
110            icon: default.icon.name().to_string(),
111            color: default.color.name().to_string(),
112            name,
113            policy: false,
114            policy_id: None,
115            extra: Map::new(),
116        });
117        next_user_context_id += 1;
118    }
119
120    identities.push(thumbnail_identity(next_user_context_id));
121    let last_user_context_id = next_user_context_id;
122    identities.push(webext_storage_local_identity());
123
124    ContainersData {
125        version: LATEST_VERSION,
126        last_user_context_id,
127        identities,
128        site_associations: Default::default(),
129        extra: Map::new(),
130    }
