5use std::collections::HashMap;
6use std::time::SystemTime;
7use sync15::{DeviceType, EngineChoicesModel};
8
9#[derive(Debug, uniffi::Record)]
10pub struct SyncParams {
11    /// Why are we performing this sync?
12    pub reason: SyncReason,
13    /// Which engines should we sync?
14    pub engines: SyncEngineSelection,
15    /// How the user's choice of engines to sync is modelled.
16    pub engine_choices_model: EngineChoicesModel,
17    /// Keys to encrypt/decrypt data from local database files.  These are
18    /// separate from the key we use to encrypt the sync payload as a whole.
19    pub local_encryption_keys: HashMap<String, String>,
20    /// Authorization for the sync server
21    pub auth_info: SyncAuthInfo,
22    /// An opaque string, as returned in the previous sync's SyncResult and
23    /// persisted to disk, or null if no such state is available. This includes
24    /// information such as the list of engines previously enabled, certain
25    /// server timestamps and GUIDs etc. If this value isn't correctly persisted
26    /// and round-tripped, each sync may look like a "first sync".
27    pub persisted_state: Option<String>,
28    /// Information about the current device, such as its name, formfactor and
29    /// FxA device ID.
30    pub device_settings: DeviceSettings,
31}

33#[derive(Debug, uniffi::Enum)]
34pub enum SyncReason {
35    Scheduled,
36    User,
37    PreSleep,
38    Startup,
39    EnabledChange,
40    Backgrounded,
41}
42
43#[derive(Debug, uniffi::Enum)]
44pub enum SyncEngineSelection {
45    All,
46    Some { engines: Vec<String> },
47}
48
49#[derive(Debug, uniffi::Record)]
50pub struct SyncAuthInfo {
51    pub kid: String,
52    pub fxa_access_token: String,
53    pub sync_key: String,
54    pub tokenserver_url: String,
55}
56
57#[derive(Debug, uniffi::Record)]
58pub struct DeviceSettings {
59    pub fxa_device_id: String,
60    pub name: String,
61    pub kind: DeviceType,
62    /// The engines this device syncs, to be published for other devices on the
63    /// account to see. Not yet published.
64    #[uniffi(default)]
65    pub enabled_engines: Vec<String>,
66}
67
68#[derive(Debug, uniffi::Record)]
69pub struct SyncResult {
70    /// Result from the sync server
71    pub status: ServiceStatus,
72    /// Engines that synced successfully
73    pub successful: Vec<String>,
74    /// Maps the names of engines that failed to sync to the reason why
75    pub failures: HashMap<String, String>,
76    /// State that should be persisted to disk and supplied to the sync method
77    /// on the next sync (See SyncParams.persisted_state).
78    pub persisted_state: String,
79    /// The list of engines which are marked as "declined" (ie, disabled) on the
80    /// sync server. The list of declined engines is global to the account
81    /// rather than to the device. Apps should use this after every sync to
82    /// update the local state (ie, to ensure that their Sync UI correctly
83    /// reflects what engines are enabled and disabled), because these could
84    /// change after every sync.
85    /// Always `None` when using `EngineChoicesModel::Device`, device's engine
86    /// choices are local without remote updates. `None` means "no change", not
87    /// "nothing declined".
88    pub declined: Option<Vec<String>>,
89    /// Earliest time that the next sync should happen at
90    pub next_sync_allowed_at: Option<SystemTime>,
91    /// JSON string encoding a `SyncTelemetryPing` object
92    pub telemetry_json: Option<String>,
93}
94
95#[derive(Debug, uniffi::Enum)]
96pub enum ServiceStatus {
97    Ok,
98    NetworkError,
99    ServiceError,
100    AuthError,
101    BackedOff,
102    OtherError,
103}
104
105impl ServiceStatus {
106    pub fn is_ok(&self) -> bool {
107        matches!(self, ServiceStatus::Ok)
108    }
