8impl Store {
9    /// Returns a bridged sync engine for addresses, for use by Desktop's Sync
10    /// framework. Constructing a `ConfigSyncEngine` only assembles structs and
11    /// never touches the DB, so this cannot fail.
12    pub fn addresses_bridged_engine(self: Arc<Self>) -> Arc<AddressesBridgedEngine> {
13        let engine = crate::sync::address::create_engine(self);
14        Arc::new(AddressesBridgedEngine::new(Box::new(engine)))
15    }

17    /// Returns a bridged sync engine for credit cards, for use by Desktop's

19    /// and never touches the DB, so this cannot fail.
20    pub fn credit_cards_bridged_engine(self: Arc<Self>) -> Arc<CreditCardsBridgedEngine> {
21        let engine = crate::sync::credit_card::create_engine(self);
22        Arc::new(CreditCardsBridgedEngine::new(Box::new(engine)))
23    }
24}

38    // Exercises the sync metadata the bridge owns: last_sync, sync_id and reset.
39    #[test]
40    fn test_sync_meta() {
41        error_support::init_for_tests();
42
43        let store = Arc::new(
44            Store::new_shared_memory("addresses-bridge", crate::db::test::test_encdec()).unwrap(),
45        );
46        let bridge = store.addresses_bridged_engine();
47
48        bridge.sync_started().unwrap();
49        // Fresh DB: never synced.
50        assert_eq!(bridge.last_sync().unwrap(), 0);
51        bridge.set_uploaded(3, vec![]).unwrap();
52        assert_eq!(bridge.last_sync().unwrap(), 3);
53
54        assert!(bridge.sync_id().unwrap().is_none());
55
56        bridge.ensure_current_sync_id("some_guid").unwrap();
57        assert_eq!(bridge.sync_id().unwrap(), Some("some_guid".to_string()));
58        // changing the sync ID resets the timestamp
59        assert_eq!(bridge.last_sync().unwrap(), 0);
60        bridge.set_uploaded(3, vec![]).unwrap();
61
62        bridge.reset_sync_id().unwrap();
63        assert_ne!(bridge.sync_id().unwrap(), Some("some_guid".to_string()));
64        assert_eq!(bridge.last_sync().unwrap(), 0);
65        bridge.set_uploaded(3, vec![]).unwrap();
66
67        // `reset` clears the guid and the timestamp.
68        bridge.reset().unwrap();
69        assert_eq!(bridge.last_sync().unwrap(), 0);
70        assert!(bridge.sync_id().unwrap().is_none());
71    }

109    // A roundtrip through the bridge's data path: stage an incoming remote
110    // address, apply it, and confirm the local-only address comes back out for
111    // upload. Unlike `test_sync_meta` this exercises the JSON (de)serialization
112    // of BSOs and the sync staging tables, mirroring the logins and tabs
113    // `test_sync_via_bridge` tests.
114    #[test]
115    fn test_sync_via_bridge() {
116        error_support::init_for_tests();
117
118        let store = Arc::new(
119            Store::new_shared_memory("addresses-bridge-roundtrip", crate::db::test::test_encdec())
120                .unwrap(),
121        );
122
123        // A local-only address: nothing on the server knows about it yet, so it
124        // should be uploaded.
125        let local = store
126            .add_address(UpdatableAddressFields {
127                name: "Local Person".to_string(),
128                street_address: "1 Local Lane".to_string(),
129                address_level2: "Seattle, WA".to_string(),
130                country: "US".to_string(),
131                ..Default::default()
132            })
133            .expect("should add local address");
134
135        let bridge = store.clone().addresses_bridged_engine();
136
137        // `sync_started` is what creates the sync staging tables.
138        bridge.sync_started().expect("should prepare for sync");
139
140        // An incoming remote address that isn't known locally. We build the
141        // envelope as raw JSON, exactly as the JS bridge hands it to us.
142        let incoming = vec![serde_json::json!({
143            "id": "remote-only-bbbb",
144            "modified": 0,
145            "payload": serde_json::json!({
146                "id": "remote-only-bbbb",
147                "entry": {
148                    "name": "Remote Person",
149                    "street-address": "99 Remote Road",
150                    "address-level2": "Portland, OR",
151                    "country": "US",
152                    "version": 1,
153                },
154            })
155            .to_string(),
156        })
157        .to_string()];
158        bridge
159            .store_incoming(incoming)
160            .expect("should store incoming");
161
162        // Applying stores the remote record locally and returns the local-only
163        // address for upload.
164        let outgoing = bridge.apply(1234).expect("should apply");
165        let changes: HashMap<String, serde_json::Value> = outgoing
166            .into_iter()
167            .map(|s| {
168                let bso: serde_json::Value = serde_json::from_str(&s).unwrap();
169                let payload: serde_json::Value =
170                    serde_json::from_str(bso["payload"].as_str().unwrap()).unwrap();
171                (payload["id"].as_str().unwrap().to_string(), payload)
172            })
173            .collect();
174
175        // Only the local address is outgoing; the just-applied remote one is not
176        // re-uploaded.
177        assert_eq!(changes.len(), 1);
178        assert_eq!(
179            changes[&local.guid]["entry"]["street-address"],
180            "1 Local Lane"
181        );
182
183        // The incoming remote address was actually persisted.
184        let stored = store
185            .get_address("remote-only-bbbb".to_string())
186            .expect("remote address should have been stored");
187        assert_eq!(stored.street_address, "99 Remote Road");
188
189        assert_eq!(bridge.last_sync().unwrap(), 1234);
190        bridge.set_uploaded(5678, vec![local.guid.clone()]).unwrap();
191        bridge.sync_finished().unwrap();
192        assert_eq!(bridge.last_sync().unwrap(), 5678);
193
194        // Acknowledging the upload cleared the record's change counter, so a
195        // subsequent sync has nothing to send.
196        assert!(bridge.apply(5678).expect("should apply again").is_empty());
197    }

199    // The credit card counterpart of `test_sync_via_bridge`. Payloads carry the
200    // card number and CVV in cleartext, while the store keeps them encrypted, so
201    // this also covers the encryption on the way in and decryption on the way
202    // out.
203    #[test]
204    fn test_credit_cards_sync_via_bridge() {
205        error_support::init_for_tests();
206
207        let store = Arc::new(
208            Store::new_shared_memory(
209                "credit-cards-bridge-roundtrip",
210                crate::db::test::test_encdec(),
211            )
212            .unwrap(),
213        );
214
215        // A local-only card: nothing on the server knows about it yet, so it
216        // should be uploaded.
217        let local = store
218            .add_credit_card(UpdatableCreditCardFields {
219                cc_name: "Local Person".to_string(),
220                cc_number: "4111111111111111".to_string(),
221                cc_cvv: Some("123".to_string()),
222                cc_exp_month: 1,
223                cc_exp_year: 2030,
224                cc_type: "visa".to_string(),
225            })
226            .expect("should add local card");
227
228        let bridge = store.clone().credit_cards_bridged_engine();
229
230        // `sync_started` is what creates the sync staging tables.
231        bridge.sync_started().expect("should prepare for sync");
232
233        // An incoming remote card that isn't known locally. We build the
234        // envelope as raw JSON, exactly as the JS bridge hands it to us.
235        let incoming = vec![serde_json::json!({
236            "id": "remote-only-cccc",
237            "modified": 0,
238            "payload": serde_json::json!({
239                "id": "remote-only-cccc",
240                "entry": {
241                    "cc-name": "Remote Person",
242                    "cc-number": "5555555555554444",
243                    "cc-cvv": "456",
244                    "cc-exp-month": 12,
245                    "cc-exp-year": 2031,
246                    "cc-type": "mastercard",
247                    "version": 3,
248                },
249            })
250            .to_string(),
251        })
252        .to_string()];
253        bridge
254            .store_incoming(incoming)
255            .expect("should store incoming");
256
257        // Applying stores the remote record locally and returns the local-only
258        // card for upload.
259        let outgoing = bridge.apply(1234).expect("should apply");
260        let changes: HashMap<String, serde_json::Value> = outgoing
261            .into_iter()
262            .map(|s| {
263                let bso: serde_json::Value = serde_json::from_str(&s).unwrap();
264                let payload: serde_json::Value =
265                    serde_json::from_str(bso["payload"].as_str().unwrap()).unwrap();
266                (payload["id"].as_str().unwrap().to_string(), payload)
267            })
268            .collect();
269
270        // Only the local card is outgoing, with its number and CVV decrypted;
271        // the just-applied remote one is not re-uploaded.
272        assert_eq!(changes.len(), 1);
273        assert_eq!(
274            changes[&local.guid]["entry"]["cc-number"],
275            "4111111111111111"
276        );
277        assert_eq!(changes[&local.guid]["entry"]["cc-cvv"], "123");
278
279        // The incoming card was persisted encrypted and reads back decrypted.
280        let stored = store
281            .get_credit_card("remote-only-cccc".to_string())
282            .expect("remote card should have been stored");
283        assert_eq!(stored.cc_number, "5555555555554444");
284        assert_eq!(stored.cc_number_last_4, "4444");
285        assert_eq!(stored.cc_cvv.as_deref(), Some("456"));
286        {
287            let db = store.lock_db().unwrap();
288            let (number_enc, cvv_enc): (String, String) = db
289                .query_row(
290                    "SELECT cc_number_enc, cc_cvv_enc FROM credit_cards_data
291                     WHERE guid = 'remote-only-cccc'",
292                    [],
293                    |row| Ok((row.get(0)?, row.get(1)?)),
294                )
295                .unwrap();
296            assert!(!number_enc.contains("5555555555554444"));
297            assert!(!cvv_enc.is_empty());
298            assert!(!cvv_enc.contains("456"));
299        }
300
301        assert_eq!(bridge.last_sync().unwrap(), 1234);
302        bridge.set_uploaded(5678, vec![local.guid.clone()]).unwrap();
303        bridge.sync_finished().unwrap();
304        assert_eq!(bridge.last_sync().unwrap(), 5678);
305
306        // Acknowledging the upload cleared the record's change counter, so a
307        // subsequent sync has nothing to send.
308        assert!(bridge.apply(5678).expect("should apply again").is_empty());
309    }
