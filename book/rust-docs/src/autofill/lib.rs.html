33/// Identifier for the autofill key, under which the key is stored in NSS.
34#[cfg(feature = "keydb")]
35static KEY_NAME: &str = "as-autofill-key";
36
37// Create a Store by passing in a db path and a static key
38//
39// Note this is only temporarily needed until a bug with UniFFI and JavaScript is fixed, which
40// prevents passing around traits in JS
41pub fn create_autofill_store_with_static_key_manager(
42    path: String,
43    key: String,
44) -> ApiResult<Arc<Store>> {
45    let encdec: ManagedEncryptorDecryptor =
46        ManagedEncryptorDecryptor::new(Arc::new(StaticKeyManager::new(key)));
47    check_canary_roundtrip(&encdec)?;
48    Ok(Arc::new(Store::new(path, Arc::new(encdec))?))
49}
50
51/// Reject a key that cannot round-trip, instead of failing on first use.
52#[handle_error(Error)]
53fn check_canary_roundtrip(encdec: &dyn EncryptorDecryptor) -> ApiResult<()> {
54    let canary = encdec.encrypt(b"tschilp".to_vec())?;
55    encdec.decrypt(canary)?;
56    Ok(())
57}

59// Create a Store with NSSKeyManager by passing in a db path and a PrimaryPasswordAuthenticator.
60//
61// Note this is only temporarily needed until a bug with UniFFI and JavaScript is fixed, which
62// prevents passing around traits in JS
63#[cfg(feature = "keydb")]
64#[uniffi::export]
65pub fn create_autofill_store_with_nss_keymanager(
66    path: String,
67    primary_password_authenticator: Arc<dyn PrimaryPasswordAuthenticator>,
68) -> ApiResult<Arc<Store>> {
69    let encdec: ManagedEncryptorDecryptor = ManagedEncryptorDecryptor::new(Arc::new(
70        NSSKeyManager::new(KEY_NAME.to_string(), primary_password_authenticator),
71    ));
72    let store = Store::new(path, Arc::new(encdec))?;
73    Ok(Arc::new(store))
74}

76#[cfg(test)]
77pub(crate) fn static_key_encryptor(key: &str) -> Result<ManagedEncryptorDecryptor> {
78    // Validate eagerly so an invalid key isn't treated as undecryptable card data.
79    jwcrypto::EncryptorDecryptor::new(key)?;
80
81    Ok(ManagedEncryptorDecryptor::new(Arc::new(
82        StaticKeyManager::new(key.to_string()),
83    )))
84}

86#[handle_error(Error)]
87pub fn create_autofill_key() -> ApiResult<String> {
88    Ok(db_crypto::create_key()?)
89}

91#[cfg(test)]
92mod tests {
93    use super::*;
94    use crate::db::models::credit_card::{decrypt_str, encrypt_str};
95    use nss_as::ensure_initialized;
96
97    #[test]
98    fn test_encrypt() {
99        ensure_initialized();
100        let ed = static_key_encryptor(&create_autofill_key().unwrap()).unwrap();
101        let cleartext = "secret";
102        let ciphertext = encrypt_str(&ed, cleartext).unwrap();
103        assert_eq!(decrypt_str(&ed, &ciphertext).unwrap(), cleartext);
104        let ed2 = static_key_encryptor(&create_autofill_key().unwrap()).unwrap();
105        assert!(matches!(
106            decrypt_str(&ed2, &ciphertext),
107            Err(Error::EncryptionError(
108                db_crypto::DbCryptoApiError::DecryptionFailed { .. }
109            ))
110        ));
111    }
112
113    #[test]
114    fn test_decryption_errors() {
115        // The shared crate maps all jwcrypto decryption failures to DecryptionFailed.
116        ensure_initialized();
117        let ed = static_key_encryptor(&create_autofill_key().unwrap()).unwrap();
118        assert!(matches!(
119            decrypt_str(&ed, "invalid-ciphertext"),
120            Err(Error::EncryptionError(
121                db_crypto::DbCryptoApiError::DecryptionFailed { .. }
122            )),
123        ));
124        assert!(matches!(
125            decrypt_str(&ed, ""),
126            Err(Error::EncryptionError(
127                db_crypto::DbCryptoApiError::DecryptionFailed { .. }
128            )),
129        ));
130    }
131
132    #[test]
133    fn test_an_invalid_key_is_rejected_up_front() {
134        ensure_initialized();
135        assert!(static_key_encryptor("not-a-key").is_err());
136    }
