98/// A credit card stored in the database, minus the encrypted fields.
99///
100/// Getting one of these never needs the encryption key, so callers listing
101/// cards by name, last four digits and expiry can do so without forcing the
102/// user to authenticate. Once they know which card they want,
103/// `Store::get_credit_card()` decrypts just that one.
104#[derive(Debug, Clone, PartialEq, Eq, Default)]
105pub struct CreditCardCandidate {
106    pub guid: String,
107    pub cc_name: String,
108    pub cc_number_last_4: String,
109    /// Whether a CVV is stored, from the ciphertext being non-empty.
110    pub has_cvv: bool,
111    pub cc_exp_month: i64,
112    pub cc_exp_year: i64,
113    pub cc_type: String,
114
115    pub time_created: i64,
116    pub time_last_used: Option<i64>,
117    pub time_last_modified: i64,
118    pub times_used: i64,
119}

121impl From<InternalCreditCard> for CreditCardCandidate {
122    fn from(card: InternalCreditCard) -> Self {
123        // The ciphertexts are dropped unread; only whether a CVV exists is kept.
124        Self {
125            guid: card.guid.to_string(),
126            cc_name: card.cc_name,
127            cc_number_last_4: card.cc_number_last_4,
128            has_cvv: !card.cc_cvv_enc.is_empty(),
129            cc_exp_month: card.cc_exp_month,
130            cc_exp_year: card.cc_exp_year,
131            cc_type: card.cc_type,
132            time_created: u64::from(card.metadata.time_created) as i64,
133            time_last_used: if card.metadata.time_last_used.0 == 0 {
134                None
135            } else {
136                Some(card.metadata.time_last_used.0 as i64)
137            },
138            time_last_modified: u64::from(card.metadata.time_last_modified) as i64,
139            times_used: card.metadata.times_used,
140        }
141    }
142}

144pub(crate) fn encrypt_str(encdec: &dyn EncryptorDecryptor, cleartext: &str) -> Result<String> {
145    let ciphertext = encdec.encrypt(cleartext.as_bytes().to_vec())?;
146    String::from_utf8(ciphertext).map_err(|e| Error::CryptoNotUtf8(format!("encrypting: {e}")))
147}
148
149pub(crate) fn decrypt_str(encdec: &dyn EncryptorDecryptor, ciphertext: &str) -> Result<String> {
150    let cleartext = encdec.decrypt(ciphertext.as_bytes().to_vec())?;
151    String::from_utf8(cleartext).map_err(|e| Error::CryptoNotUtf8(format!("decrypting: {e}")))
152}
153
154pub(crate) fn encrypt_optional_str(
155    encdec: &dyn EncryptorDecryptor,
156    cleartext: Option<&str>,
157) -> Result<String> {
158    match cleartext {
159        Some(value) if !value.is_empty() => encrypt_str(encdec, value),
160        _ => Ok(String::new()),
161    }
162}
163
164pub(crate) fn decrypt_optional_str(
165    encdec: &dyn EncryptorDecryptor,
166    ciphertext: &str,
167) -> Result<Option<String>> {
168    if ciphertext.is_empty() {
169        Ok(None)
170    } else {
171        Ok(Some(decrypt_str(encdec, ciphertext)?))
172    }
173}

175// Wow - strings are hard! we need the last 4 chars of a string.
176pub(crate) fn get_last_4(v: &str) -> String {
177    v.chars()
178        .rev()
179        .take(4)
180        .collect::<Vec<_>>()
181        .into_iter()
182        .rev()
183        .collect::<String>()
184}

186// This is used to "externalize" a credit-card, suitable for handing back to
187// consumers.
188impl InternalCreditCard {
189    pub(crate) fn into_external(self, encdec: &dyn EncryptorDecryptor) -> Result<CreditCard> {
190        // A scrubbed card has an empty ciphertext and stays empty; anything
191        // else the key cannot read fails the read.
192        let cc_number = if self.cc_number_enc.is_empty() {
193            String::new()
194        } else {
195            decrypt_str(encdec, &self.cc_number_enc)?
196        };
197        let cc_cvv = decrypt_optional_str(encdec, &self.cc_cvv_enc)?;
198        Ok(CreditCard {
199            guid: self.guid.to_string(),
200            cc_name: self.cc_name,
201            cc_number,
202            cc_cvv,
203            cc_number_last_4: self.cc_number_last_4,
204            cc_exp_month: self.cc_exp_month,
205            cc_exp_year: self.cc_exp_year,
206            cc_type: self.cc_type,
207            // note we can't use u64 in uniffi
208            time_created: u64::from(self.metadata.time_created) as i64,
209            time_last_used: if self.metadata.time_last_used.0 == 0 {
210                None
211            } else {
212                Some(self.metadata.time_last_used.0 as i64)
213            },
214            time_last_modified: u64::from(self.metadata.time_last_modified) as i64,
215            times_used: self.metadata.times_used,
216        })
217    }
218}

220// NOTE: No `PartialEq` here because the same card number will encrypt to a
221// different value each time it is encrypted, making it meaningless to compare.
222#[derive(Debug, Clone, Default)]
223pub struct InternalCreditCard {
224    pub guid: Guid,
225    pub cc_name: String,
226    pub cc_number_enc: String,
227    /// Empty when no CVV is stored, like a scrubbed number.
228    pub cc_cvv_enc: String,
229    pub cc_number_last_4: String,
230    pub cc_exp_month: i64,
231    pub cc_exp_year: i64,
232    // Credit card types are a fixed set of strings as defined in the link below
233    // (https://searchfox.org/mozilla-central/rev/7ef5cefd0468b8f509efe38e0212de2398f4c8b3/toolkit/modules/CreditCard.jsm#9-22)
234    pub cc_type: String,
235    pub metadata: Metadata,
236}

238impl InternalCreditCard {
239    pub fn from_row(row: &Row<'_>) -> std::result::Result<InternalCreditCard, rusqlite::Error> {
240        Ok(Self {
241            guid: Guid::from_string(row.get("guid")?),
242            cc_name: row.get("cc_name")?,
243            cc_number_enc: row.get("cc_number_enc")?,
244            cc_cvv_enc: row.get("cc_cvv_enc")?,
245            cc_number_last_4: row.get("cc_number_last_4")?,
246            cc_exp_month: row.get("cc_exp_month")?,
247            cc_exp_year: row.get("cc_exp_year")?,
248            cc_type: row.get("cc_type")?,
249            metadata: Metadata {
250                time_created: row.get::<_, Timestamp>("time_created")?.sanitized(),
251                time_last_used: row.get::<_, Timestamp>("time_last_used")?.sanitized(),
252                time_last_modified: row.get::<_, Timestamp>("time_last_modified")?.sanitized(),
253                times_used: row.get("times_used")?,
254                sync_change_counter: row.get("sync_change_counter")?,
255            },
256        })
257    }
258
259    pub fn has_scrubbed_data(&self) -> bool {
260        self.cc_number_enc.is_empty()
261    }
