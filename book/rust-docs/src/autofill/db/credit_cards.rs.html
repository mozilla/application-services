29/// An empty `cc_number` is stored as an empty `cc_number_enc` rather than as
30/// a ciphertext of "": a card may be saved without a number, and the empty
31/// ciphertext is what marks a scrubbed card - both for reads and for the
32/// Sync upload filter, which runs on the ciphertext.
33fn encrypt_number(encdec: &dyn EncryptorDecryptor, cc_number: &str) -> Result<String> {
34    if cc_number.is_empty() {
35        Ok(String::new())
36    } else {
37        encrypt_str(encdec, cc_number)
38    }
39}

41pub(crate) fn add_credit_card(
42    conn: &Connection,
43    encdec: &dyn EncryptorDecryptor,
44    new_credit_card_fields: UpdatableCreditCardFields,
45) -> Result<InternalCreditCard> {
46    let now = Timestamp::now();
47
48    // We return an InternalCreditCard, so set it up first, including the
49    // missing fields, before we insert it.
50    let credit_card = InternalCreditCard {
51        guid: Guid::random(),
52        cc_name: new_credit_card_fields.cc_name,
53        cc_number_enc: encrypt_number(encdec, &new_credit_card_fields.cc_number)?,
54        cc_cvv_enc: encrypt_optional_str(encdec, new_credit_card_fields.cc_cvv.as_deref())?,
55        cc_number_last_4: get_last_4(&new_credit_card_fields.cc_number),
56        cc_exp_month: new_credit_card_fields.cc_exp_month,
57        cc_exp_year: new_credit_card_fields.cc_exp_year,
58        // Credit card types are a fixed set of strings as defined in the link below
59        // (https://searchfox.org/mozilla-central/rev/7ef5cefd0468b8f509efe38e0212de2398f4c8b3/toolkit/modules/CreditCard.jsm#9-22)
60        cc_type: new_credit_card_fields.cc_type,
61        metadata: Metadata {
62            time_created: now,
63            time_last_modified: now,
64            ..Default::default()
65        },
66    };
67
68    let tx = conn.unchecked_transaction()?;
69    add_internal_credit_card(&tx, &credit_card)?;
70    tx.commit()?;
71    Ok(credit_card)
72}

74/// Adds a credit card **including metadata**, taking the guid, timestamps and
75/// sync change counter from the caller rather than generating them. Normally you
76/// will use `add_credit_card` instead; this is for importing records from
77/// another store that already have metadata.
78///
79/// The number is supplied in cleartext and encrypted here, matching
80/// `add_credit_card`.
81pub(crate) fn add_credit_card_with_meta(
82    conn: &Connection,
83    encdec: &dyn EncryptorDecryptor,
84    fields: UpdatableCreditCardFields,
85    meta: CreditCardMeta,
86) -> Result<InternalCreditCard> {
87    let tx = conn.unchecked_transaction()?;
88    let card = internal_credit_card_from_meta(encdec, fields, &meta)?;
89    add_internal_credit_card(&tx, &card)?;
90    tx.commit()?;
91    Ok(card)
92}

94/// Adds multiple credit cards **including metadata** within a single
95/// transaction. Each record gets its own result, so a record that fails to
96/// insert is reported as `Err(message)` without aborting the rest of the batch.
97pub(crate) fn add_many_credit_cards_with_meta(
98    conn: &Connection,
99    encdec: &dyn EncryptorDecryptor,
100    entries: Vec<UpdatableCreditCardFieldsWithMeta>,
101) -> Result<Vec<std::result::Result<InternalCreditCard, String>>> {
102    let tx = conn.unchecked_transaction()?;
103    let mut results = Vec::with_capacity(entries.len());
104    for entry in entries {
105        let card = match internal_credit_card_from_meta(encdec, entry.fields, &entry.meta) {
106            Ok(card) => card,
107            Err(e) => {
108                results.push(Err(e.to_string()));
109                continue;
110            }
111        };
112        match with_savepoint(&tx, || add_internal_credit_card(&tx, &card))? {
113            Ok(()) => results.push(Ok(card)),
114            Err(e) => results.push(Err(e.to_string())),
115        }
116    }
117    tx.commit()?;
118    Ok(results)
119}

121/// Removes every credit card and every credit card tombstone, in one
122/// transaction.
123///
124/// Deleting the rows alone is not enough. A delete leaves a tombstone behind for
125/// any guid the sync mirror knows, and the insert trigger then rejects re-adding
126/// that guid, so a wipe that kept them could not be followed by a re-import of
127/// the same records. Clearing both tables is what makes the wipe repeatable.
128pub(crate) fn delete_all_credit_cards(conn: &Connection) -> Result<()> {
129    let tx = conn.unchecked_transaction()?;
130    tx.execute("DELETE FROM credit_cards_data", [])?;
131    // After the data, so the tombstones the delete trigger just created go too.
132    tx.execute("DELETE FROM credit_cards_tombstones", [])?;
133    tx.commit()?;
134    Ok(())
135}

137/// Adds tombstones for records that were deleted locally but not yet uploaded,
138/// within a single transaction and with a result per record. `time_deleted` comes
139/// from the caller rather than being stamped as now, so that a deletion imported
140/// from another store keeps its original time. Without the tombstone the next
141/// sync has nothing to say the record was deleted and takes the server copy.
142pub(crate) fn add_many_credit_card_tombstones(
143    conn: &Connection,
144    tombstones: Vec<(String, i64)>,
145) -> Result<Vec<std::result::Result<String, String>>> {
146    let tx = conn.unchecked_transaction()?;
147    let mut results = Vec::with_capacity(tombstones.len());
148    for (guid, time_deleted) in tombstones {
149        let inserted = with_savepoint(&tx, || {
150            tx.execute(
151                "INSERT INTO credit_cards_tombstones (guid, time_deleted)
152                 VALUES (:guid, :time_deleted)",
153                rusqlite::named_params! {
154                    ":guid": &guid,
155                    ":time_deleted": timestamp_from_millis(time_deleted),
156                },
157            )?;
158            Ok(())
159        })?;
160        match inserted {
161            Ok(()) => results.push(Ok(guid)),
162            Err(e) => results.push(Err(e.to_string())),
163        }
164    }
165    tx.commit()?;
166    Ok(results)
167}

169fn internal_credit_card_from_meta(
170    encdec: &dyn EncryptorDecryptor,

186            time_last_modified: timestamp_from_millis(meta.time_last_modified),

193/// Updates a credit card **including metadata**, setting both its fields and its
194/// timestamps and `times_used` to the supplied values. Normally you will use
195/// `update_credit_card` instead, which owns the metadata itself; this is for
196/// keeping a record identical to one held in another store. Errors with

212    tx.commit()?;
213    Ok(())

293/// Replaces the user-editable fields of `guid`. A `cc_cvv` of `None` keeps
294/// the stored CVV and `Some("")` clears it, so a caller that does not handle
295/// CVVs cannot erase one.
296pub fn update_credit_card(
297    conn: &Connection,
298    encdec: &dyn EncryptorDecryptor,
299    guid: &Guid,
300    credit_card: &UpdatableCreditCardFields,
301) -> Result<()> {
302    let cc_number_enc = encrypt_number(encdec, &credit_card.cc_number)?;
303    let cc_cvv_enc = match credit_card.cc_cvv.as_deref() {
304        // None binds as NULL, which the COALESCE below turns into "keep".
305        None => None,
306        Some(cc_cvv) => Some(encrypt_optional_str(encdec, Some(cc_cvv))?),
307    };
308    let tx = conn.unchecked_transaction()?;
309    tx.execute(
310        "UPDATE credit_cards_data
311        SET cc_name                     = :cc_name,
312            cc_number_enc               = :cc_number_enc,
313            cc_cvv_enc                  = COALESCE(:cc_cvv_enc, cc_cvv_enc),
314            cc_number_last_4            = :cc_number_last_4,
315            cc_exp_month                = :cc_exp_month,
316            cc_exp_year                 = :cc_exp_year,
317            cc_type                     = :cc_type,
318            time_last_modified          = :time_last_modified,
319            sync_change_counter         = sync_change_counter + 1
320        WHERE guid                      = :guid",
321        rusqlite::named_params! {
322            ":cc_name": credit_card.cc_name,
323            ":cc_number_enc": cc_number_enc,
324            ":cc_cvv_enc": cc_cvv_enc,
325            ":cc_number_last_4": get_last_4(&credit_card.cc_number),
326            ":cc_exp_month": credit_card.cc_exp_month,
327            ":cc_exp_year": credit_card.cc_exp_year,
328            ":cc_type": credit_card.cc_type,
329            ":time_last_modified": Timestamp::now(),
330            ":guid": guid,
331        },
332    )?;
333
334    tx.commit()?;
335    Ok(())
336}

338/// Updates all fields including metadata - although the change counter gets
339/// slightly special treatment, see `CounterUpdate`.
340pub(crate) fn update_internal_credit_card(
341    tx: &Transaction<'_>,
342    card: &InternalCreditCard,
343    counter: CounterUpdate,
344) -> Result<()> {
345    let (counter_sql, counter_value) = counter.as_sql();
346    let rows_changed = tx.execute(
347        &format!(
348            "UPDATE credit_cards_data
349        SET cc_name                     = :cc_name,
350            cc_number_enc               = :cc_number_enc,
351            cc_cvv_enc                  = :cc_cvv_enc,
352            cc_number_last_4            = :cc_number_last_4,
353            cc_exp_month                = :cc_exp_month,
354            cc_exp_year                 = :cc_exp_year,
355            cc_type                     = :cc_type,
356            time_created                = :time_created,
357            time_last_used              = :time_last_used,
358            time_last_modified          = :time_last_modified,
359            times_used                  = :times_used,
360            sync_change_counter         = {counter_sql}
361        WHERE guid                      = :guid"
362        ),
363        rusqlite::named_params! {
364            ":cc_name": card.cc_name,
365            ":cc_number_enc": card.cc_number_enc,
366            ":cc_cvv_enc": card.cc_cvv_enc,
367            ":cc_number_last_4": card.cc_number_last_4,
368            ":cc_exp_month": card.cc_exp_month,
369            ":cc_exp_year": card.cc_exp_year,
370            ":cc_type": card.cc_type,
371            ":time_created": card.metadata.time_created,
372            ":time_last_used": card.metadata.time_last_used,
373            ":time_last_modified": card.metadata.time_last_modified,
374            ":times_used": card.metadata.times_used,
375            ":counter": counter_value,
376            ":guid": card.guid,
377        },
378    )?;
379    if rows_changed == 0 {
380        return Err(Error::NoSuchRecord(card.guid.to_string()));
381    }
382    Ok(())
383}

411/// Scrub cards the store's encryptor cannot decrypt, so Sync replaces them
412/// from the server. This deletes on *any* decryption failure, permanent or
413/// not - with an NSS-backed key, a cancelled primary-password prompt fails
414/// decryption too and would scrub every card. Only call this where the key
415/// is known usable; today's only caller is iOS, whose key is static.
416pub fn scrub_undecryptable_credit_card_data_for_remote_replacement(
417    db: &AutofillDb,
418) -> Result<CreditCardsDeletionMetrics> {
419    let conn = &db.writer;
420    let tx = conn.unchecked_transaction()?;
421    let mut scrubbed_records = 0;
422
423    let undecryptable_record_ids = get_all_credit_cards(conn)?
424        .into_iter()
425        .filter(|credit_card| {
426            db.encdec
427                .decrypt(credit_card.cc_number_enc.as_bytes().to_vec())
428                .is_err()
429        })
430        .map(|credit_card| credit_card.guid)
431        .collect::<Vec<_>>();
432
433    // Reset the cc_number_enc field as well as the meta fields of the record so if the record was previously synced
434    // it will be overwritten
435    sql_support::each_chunk(&undecryptable_record_ids, |chunk, _| -> Result<()> {
436        let scrubbed = tx.execute(
437            &format!(
438                "UPDATE credit_cards_data
439                SET cc_number_enc = '',
440                    cc_cvv_enc = '',
441                    time_created = 0,
442                    time_last_used = 0,
443                    time_last_modified = 0,
444                    times_used = 0,
445                    sync_change_counter = 0
446                WHERE guid IN ({})",
447                sql_support::repeat_sql_values(chunk.len())
448            ),
449            rusqlite::params_from_iter(chunk),
450        )?;
451        scrubbed_records += scrubbed;
452        Ok(())
453    })?;
454
455    tx.commit()?;
456    Ok(CreditCardsDeletionMetrics {
457        total_scrubbed_records: scrubbed_records as u64,
458    })
459}
